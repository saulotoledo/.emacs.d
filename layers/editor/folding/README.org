#+CATEGORY: editor

* Description
Provides automatic Emacs treesit grammar management and code folding for
programming languages. Uses treesit-auto to redirect major modes to their
Tree-sitter-backed (-ts-mode) counterparts, and treesit-fold for syntax-aware
code folding with fringe indicators.

* Emacs Configuration
** treesit-auto: Automatic Grammar Management
Emacs' built-in ~treesit~ API integrates the Tree-sitter parser runtime and
language grammars. Emacs 29+ provides ~treesit~ and the corresponding
~*-ts-mode~ modes for fast syntax highlighting and structural editing;
~treesit-auto~ manages their mode remapping and grammar installation.

Remember to upgrade the versions when possible. New versions might not be
compatible with your current Emacs build.

** treesit-fold: Syntax-Aware Code Folding
~treesit-fold~ is a syntax-aware folding engine built for the Treesit system in
Emacs 29+. Unlike traditional folding that relies on indentation "guessing," it
uses a concrete parser to identify the precise boundaries of YAML blocks, maps,
and lists. This allows for lightning-fast, reliable collapsing of massive
datasets, often enhanced by visual indicators in the margin that let you
navigate complex hierarchies with the same ease as an Org-mode document.

#+begin_src emacs-lisp
;; --- Performance: Idle-timer debounce for fringe indicators ---
;;
;; Problem: treesit-fold-indicators--post-command runs synchronously on
;; *every* keystroke via post-command-hook. While typing broken/incomplete
;; code (common in TS/Next.js), the Tree-sitter parser repeatedly rebuilds
;; the CST and Emacs re-renders fringe markers, causing ~11 % CPU usage and
;; input lag (see profiler reports).
;;
;; Fix: Replace the synchronous call with a buffer-local idle timer.
;; Each keystroke cancels the pending timer and schedules a fresh one.
;; The real update only fires after the user pauses for 0.5 s, meaning
;; *zero* indicator overhead while actively typing.
(defvar-local cfg-editor/treesit-fold-idle-timer nil
  "Buffer-local idle timer for deferred treesit-fold indicator updates.")

(define-advice treesit-fold-indicators--post-command
    (:around (orig-fn &rest args) idle-debounce)
  "Defer indicator updates to idle time; cancel-on-type for zero typing overhead.
Skip entirely when indicators are not active in this buffer."
  ;; Fast-path: avoid timer churn in buffers where indicators aren't on
  ;; (minibuffer, org-mode, etc.).
  (when (bound-and-true-p treesit-fold-indicators-mode)
    (when (timerp cfg-editor/treesit-fold-idle-timer)
      (cancel-timer cfg-editor/treesit-fold-idle-timer))
    (let ((buf (current-buffer)))
      (setq cfg-editor/treesit-fold-idle-timer
            (run-with-idle-timer 0.8 nil
                                 (lambda ()
                                   (when (buffer-live-p buf)
                                     (with-current-buffer buf
                                       (ignore-errors (apply orig-fn args))))))))))

(defun cfg-editor/treesit-fold-toggle-all ()
  "Toggle all folds in the current buffer."
  (interactive)
  (if (bound-and-true-p cfg-editor/treesit-folded-p)
      (progn (treesit-fold-open-all) (setq-local cfg-editor/treesit-folded-p nil))
    (progn (treesit-fold-close-all) (setq-local cfg-editor/treesit-folded-p t))))

(use-package treesit-fold
  :pin "JCS-ELPA"
  :config
  ;; Enable indicators globally
  (global-treesit-fold-indicators-mode 1)

  ;; Enable folding for all programming modes
  (add-hook 'prog-mode-hook #'treesit-fold-mode))
#+end_src

** Tree-sitter Import Folding
Import folding is implemented here, while each language layer registers its own
Tree-sitter node names. This keeps the folding algorithm shared without mixing
language-specific grammar details into this layer.

#+begin_src emacs-lisp
(declare-function treesit-fold--cons-add "treesit-fold" (&rest ranges))
(declare-function treesit-fold-close "treesit-fold" (&optional node))
(declare-function treesit-fold--format-overlay-text "treesit-fold" (beg end))
(declare-function treesit-fold-usable-mode-p "treesit-fold" (&optional mode))
(defvar treesit-fold-range-alist)

(defvar cfg-editor/treesit-fold-import-rules nil
  "Alist of (MAJOR-MODE . IMPORT-NODE-TYPES) registrations.
IMPORT-NODE-TYPES contains Tree-sitter node type strings.")

(defconst cfg-editor/treesit-fold-comment-node-types
  '("line_comment" "block_comment" "comment")
  "Tree-sitter node types allowed between adjacent import nodes.")

(defun cfg-editor--treesit-fold-import-range (node offset)
  "Return a fold range for the import run beginning at NODE and OFFSET."
  (let* ((import-types (alist-get major-mode cfg-editor/treesit-fold-import-rules))
         (last-node node)
         (sibling (treesit-node-next-sibling node t)))
    (when (member (treesit-node-type node) import-types)
      (while (and sibling
                  (let ((type (treesit-node-type sibling)))
                    (or (member type import-types)
                        (member type cfg-editor/treesit-fold-comment-node-types))))
        (when (member (treesit-node-type sibling) import-types)
          (setq last-node sibling))
        (setq sibling (treesit-node-next-sibling sibling t)))
      (unless (treesit-node-eq last-node node)
        (treesit-fold--cons-add
         (cons (treesit-node-start node) (treesit-node-end last-node))
         offset)))))

(defun cfg-editor--install-treesit-fold-import-rules ()
  "Install registered import ranges into `treesit-fold-range-alist'."
  (when (boundp 'treesit-fold-range-alist)
    (dolist (registration cfg-editor/treesit-fold-import-rules)
      (let* ((mode (car registration))
             (node-types (cdr registration))
             (mode-ranges (alist-get mode treesit-fold-range-alist)))
        (dolist (node-type node-types)
          (setf (alist-get (intern node-type) mode-ranges)
                #'cfg-editor--treesit-fold-import-range))
        (setf (alist-get mode treesit-fold-range-alist) mode-ranges)))))

(defun cfg-editor/register-treesit-fold-imports (mode node-types)
  "Register NODE-TYPES as import folds for MODE.
MODE is a major-mode symbol and NODE-TYPES are Tree-sitter type strings."
  (setf (alist-get mode cfg-editor/treesit-fold-import-rules)
        (delete-dups (copy-sequence node-types)))
  (when (featurep 'treesit-fold)
    (cfg-editor--install-treesit-fold-import-rules)))

(defun cfg-editor/treesit-fold-close-imports ()
  "Fold the leading import run in the current Tree-sitter buffer."
  (when (and (bound-and-true-p treesit-fold-mode)
             (treesit-fold-usable-mode-p)
             (treesit-parser-list))
    (let* ((import-types (alist-get major-mode cfg-editor/treesit-fold-import-rules))
           (root (treesit-buffer-root-node))
           (first-import
            (seq-find (lambda (node)
                        (member (treesit-node-type node) import-types))
                      (treesit-node-children root))))
      (when first-import
        (treesit-fold-close first-import)))))

(defun cfg-editor--treesit-fold-import-node-type (beg end)
  "Return the registered import node type for the range BEG to END."
  (when-let* ((import-types (alist-get major-mode cfg-editor/treesit-fold-import-rules))
              (root (treesit-buffer-root-node))
              (node (seq-find
                     (lambda (candidate)
                       (and (member (treesit-node-type candidate) import-types)
                            (= beg (treesit-node-start candidate))))
                     (treesit-node-children root)))
              (range (cfg-editor--treesit-fold-import-range node '(0 . 0))))
    (when (= end (cdr range))
      (treesit-node-type node))))

(defun cfg-editor--treesit-fold-format-overlay-text (orig-fn beg end)
  "Use an import label for the fold from BEG to END, or call ORIG-FN."
  (if-let ((node-type (cfg-editor--treesit-fold-import-node-type beg end)))
      (format " %s..."
              (replace-regexp-in-string "_" " " node-type))
    (funcall orig-fn beg end)))

(with-eval-after-load 'treesit-fold
  (cfg-editor--install-treesit-fold-import-rules)
  (add-hook 'treesit-fold-mode-hook #'cfg-editor/treesit-fold-close-imports)
  (advice-add 'treesit-fold--format-overlay-text
              :around #'cfg-editor--treesit-fold-format-overlay-text))
#+end_src
