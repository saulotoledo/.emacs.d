#+TITLE: Regions Layer
#+CATEGORY: editor
#+PROPERTY: header-args:emacs-lisp :tangle yes

* Description
Provides semantic region selection and expansion tools for Emacs. Expand region
incrementally increases the selected region by syntactic and semantic units
(words, symbols, quotes, code blocks, paragraphs) with performance optimizations
for large Org-mode files.

* Emacs Configuration
** Expand Region
#+begin_src emacs-lisp
(defvar er/try-expand-list)
(defvar er/save-mode-excursion)

(defun cfg-editor/er-mark-org-code-block-fast ()
  "Mark an org-code-block, searching at most 10KB to prevent lag."
  (interactive)
  (let ((case-fold-search t)
        (re "#\\+begin_\\(\\sw+\\)")
        (limit (max (point-min) (- (point) 10000))))
    (when (or (looking-at re)
              (search-backward-regexp re limit t))
      (set-mark (point))
      (search-forward (concat "#+end_" (match-string 1))
                      (min (point-max) (+ (point) 20000)) t)
      (exchange-point-and-mark))))

(defun cfg-editor/er-expand-region-large-buffer-advice (orig-fun &rest args)
  "Dynamically override `er/try-expand-list' in large Org buffers.
ORIG-FUN is the advised function. ARGS are its arguments."
  (if (and (derived-mode-p 'org-mode)
           (> (buffer-size) 50000))
      (let ((er/try-expand-list
             '(er/mark-word
               er/mark-symbol
               er/mark-sentence
               er/mark-paragraph
               cfg-editor/er-mark-org-code-block-fast))
            (er/save-mode-excursion nil))
        (message "Expand-region: Optimized for large buffer.")
        (apply orig-fun args))
    (apply orig-fun args)))

(use-package expand-region
  :custom
  ;; Global optimization: disabling state-save prevents UI lag in large buffers
  (er/save-state-before-expand nil)
  (er/history-is-kept-at-point nil)
  :config
  (advice-add 'er/expand-region :around #'cfg-editor/er-expand-region-large-buffer-advice)
  (advice-add 'er/contract-region :around #'cfg-editor/er-expand-region-large-buffer-advice))
#+end_src

** Multiple Cursors
Multiple Cursors lets you edit many places at once: mark the next or
previous occurrence and type to edit them all simultaneously. Below are
a couple of convenient keybindings.

#+begin_src emacs-lisp
(declare-function doom-modeline-def-segment "doom-modeline")
(declare-function mc/load-lists "multiple-cursors-core")
(declare-function mc/save-lists "multiple-cursors-core")
(defvar cfg-editor/mc-cursor-count nil
  "Doom Modeline segment state for the multiple-cursors count.")

                 er/mark-inside-quotes

                 er/mark-word

                 exchange-point-and-mark

                 forward-word
                 indent-for-tab-command
                 just-one-space
                 kill-line
                 kill-region
                 mark-word
                 move-beginning-of-line
                 move-end-of-line
                 mwim-beginning-of-code-or-line
                 mwim-beginning-of-line
                 mwim-end-of-code-or-line
                 mwim-end-of-line

(declare-function mc/num-cursors "multiple-cursors")
(declare-function cfg-editor/mc-count-cursors "multiple-cursors")
(with-eval-after-load 'multiple-cursors
  (defun cfg-editor/mc-count-cursors ()
    "Return the number of active cursors for the mode line."
    (when (bound-and-true-p multiple-cursors-mode)
      (format " MC:%d" (mc/num-cursors))))

  (with-eval-after-load 'doom-modeline
    (doom-modeline-def-segment cfg-editor/mc-cursor-count
      "Display the number of active cursors in multiple-cursors mode."
      (cfg-editor/mc-count-cursors))))
