* Description
This layer provides enhanced file management capabilities within Emacs by
configuring built-in Dired, Dirvish, and diredfl. It enables modern directory
navigation with subtree previews, theme-adaptive file attribute highlighting,
and integrated searching via ripgrep.

(defun cfg-navigation/refresh-dired-and-children (&rest _)
  "Force-refresh the current Dired buffer and any related child buffers.
It is used after wdired and completed file operations to make changes
propagate to related buffers and to rebuild Dirvish subtrees."
  (let ((parent-dir (expand-file-name default-directory)))
    (dolist (buf (buffer-list))
      (with-current-buffer buf
        (let ((buffer-dir (expand-file-name default-directory)))
          (when (and (derived-mode-p 'dired-mode)
                     (condition-case nil
                         (or (file-equal-p buffer-dir parent-dir)
                             (file-in-directory-p buffer-dir parent-dir))
                       (file-error nil)))
            (revert-buffer t t)))))))

(defun cfg-navigation/dired-refresh-after-deletion (orig-fun deletions &rest args)
  "Refresh Dirvish after ORIG-FUN with ARGS successfully deletes an entry.
DELETIONS is Dired's list of file/marker pairs. Refresh only when a
target disappeared, so canceled or wholly failed deletion operations do
not disturb the current buffer."
  (let* ((targets (mapcar #'car deletions))
         (existing-targets
          (mapcar (lambda (target) (cons target (file-exists-p target)))
                  targets))
        result)
    (setq result (apply orig-fun deletions args))
    (when (and (derived-mode-p 'dired-mode)
               (fboundp 'dirvish-curr)
               (ignore-errors (dirvish-curr))
               (catch 'deleted
                 (dolist (target targets)
                   (when (and (cdr (assoc target existing-targets))
                              (not (file-exists-p target)))
                     (throw 'deleted t)))
                 nil))
      (cfg-navigation/refresh-dired-and-children))
    result))

(defvar cfg-navigation/dired-last-created nil
  "Cons of (buffer . path) recorded before file/dir creation.
Consumed by `cfg-navigation/dired-goto-last-created' via
`dirvish-after-revert-hook'.")

(defun cfg-navigation/dired-record-created (path &rest _)
  "Before creating PATH, save it for post-revert cursor repositioning."
  (setq cfg-navigation/dired-last-created
        (cons (current-buffer)
              (expand-file-name (directory-file-name path)))))

(defun cfg-navigation/dired-goto-last-created ()
  "Jump to the last-created Dired entry after Dirvish reverts.
Appended to `dirvish-after-revert-hook' so it runs after
`dirvish-subtree--revert' has already re-expanded all subtrees.

`dired-goto-file' (Emacs 30) searches text via `dired-subdir-alist', which
Dirvish subtrees do not populate. We therefore fall back to
`dirvish-subtree-expand-to', which navigates the overlay tree correctly."
  (when (and cfg-navigation/dired-last-created
             (eq (current-buffer) (car cfg-navigation/dired-last-created)))
    (let ((target (cdr cfg-navigation/dired-last-created)))
      (setq cfg-navigation/dired-last-created nil)
      (or (dired-goto-file target)
          (when (fboundp 'dirvish-subtree-expand-to)
            (condition-case nil
                (dirvish-subtree-expand-to target)
              (user-error nil)))))))

(defun cfg-navigation/dirvish-redisplay-after-revert ()
  "Re-render Dirvish overlays after a background Dired revert.
Dirvish icons are overlays, and its normal redisplay hook runs only when
the Dired window is selected. Auto-Revert can update an unselected Dired
buffer, so explicitly refresh the Dirvish overlays here."
  (when (and (derived-mode-p 'dired-mode)
             (fboundp 'dirvish-curr)
             (dirvish-curr)
             (fboundp 'dirvish--redisplay))
    (with-demoted-errors "Dirvish redisplay: %S"
      (dirvish--redisplay))))

(use-package dired
  :ensure nil ; Built-in
  :custom
  ;; IMPORTANT: We need to add "-l" below or it will fail in WSL2 due to
  ;; missing //DIRED// in the output of ls, required by dired to parse the
  ;; output correctly.
  ;; NOTE FOR MAC USERS: If you are on Mac, make sure to use GNU ls (e.g.,
  ;; via coreutils) and not the BSD ls that comes with macOS, as the latter
  ;; may not support the required options for dired to work properly.
  (dired-listing-switches "-ahl --group-directories-first")
  (dired-clean-confirm-killing-deleted-buffers nil) ; No confirmation prompt
  :config
  (advice-add 'wdired-finish-edit :after #'cfg-navigation/refresh-dired-and-children)
  (advice-add 'dired-internal-do-deletions :around
              #'cfg-navigation/dired-refresh-after-deletion)

  ;; After creating a file/dir inside a Dirvish subtree, the cursor jumps to
  ;; the tree root instead of the new entry. Root cause: Dirvish's own
  ;; :after advice on `dired-create-directory' calls `revert-buffer', which
  ;; invokes `dirvish-revert'. That function snapshots the *current* cursor
  ;; position as :old-index (which is wrong because `dired-add-file' silently
  ;; failed for the empty-subtree case), reverts the buffer, then
  ;; `dirvish-subtree--revert' re-expands all subtrees and navigates back to
  ;; that wrong :old-index.
  ;;
  ;; Fix: record the intended target *before* any of that happens, then use
  ;; `dirvish-after-revert-hook' (appended so it runs *after*
  ;; `dirvish-subtree--revert' has already re-inserted the listing) to jump
  ;; to the new entry.
  ;;
  ;; Navigation note: Emacs 30's `dired-goto-file' uses a text search against
  ;; `dired-subdir-alist', which Dirvish subtrees never populate (they use
  ;; overlays instead). So we fall back to `dirvish-subtree-expand-to'
  ;; which navigates via the overlay structure and correctly finds entries
  ;; inside subtrees.
  (advice-add 'dired-create-directory  :before #'cfg-navigation/dired-record-created)
  (advice-add 'dired-create-empty-file :before #'cfg-navigation/dired-record-created)
  ;; t = append, so we run after dirvish-subtree--revert (which is prepended)
  (add-hook 'dirvish-after-revert-hook #'cfg-navigation/dired-goto-last-created t)
  ;; Icons are overlays and must be rendered again when Auto-Revert refreshes
  ;; an unselected Dired buffer.
  (add-hook 'dirvish-after-revert-hook #'cfg-navigation/dirvish-redisplay-after-revert t)

  (define-key dired-mode-map (kbd "C-<up>") #'dired-up-directory)
  (define-key dired-mode-map (kbd "M-n") #'dired-create-empty-file)
  (add-hook 'dired-mode-hook (lambda () (setq-local line-spacing 0.1)))
  (global-auto-revert-mode 1)
  ;; Do NOT set auto-revert-check-vc-info to t: the auto-revert timer fires
  ;; periodically even during normal editing (no external file changes) and
  ;; with that flag set it calls vc-refresh-state -> vc-registered -> heavy
  ;; filesystem checks on every tick. diff-hl and magit already keep VC state
  ;; current, so this polling is redundant and wastes ~4% of CPU.
  (setq auto-revert-check-vc-info nil)
  (setq dired-auto-revert-buffer t))

;; Register dired commands that should use literal text entry (not matched candidates)
;; when Ivy is active. This allows copying/renaming to new names like .env.example -> .env.
(with-eval-after-load 'ivy
  (when (fboundp 'cfg-completion/ivy-add-literal-done-callers)
    (cfg-completion/ivy-add-literal-done-callers
     'dired-do-copy
     'dired-do-rename
     'dired-create-directory)))

#+begin_src emacs-lisp
(declare-function dirvish-subtree--depth "dirvish")
(declare-function dirvish-subtree--expanded-p "dirvish")
(declare-function dirvish--kill-buffer "dirvish")
(declare-function dirvish--run-with-delay "dirvish")
(declare-function dirvish--preview-update "dirvish")
(declare-function cfg-navigation/dirvish-subtree-direct-children nil)
(declare-function cfg-navigation/dirvish-subtree-only-child-directory nil)
(declare-function cfg-navigation/dirvish-refresh-preview-cache nil)
(declare-function cfg-navigation/dirvish-invalidate-saved-preview nil)

(use-package dirvish
  :bind (:map dirvish-mode-map
              ("TAB"       . dirvish-subtree-toggle)
              ("<right>"   . cfg-navigation/dirvish-subtree-right)
              ("C-<right>" . cfg-navigation/dirvish-subtree-expand-chain)
              ("<left>"    . cfg-navigation/dirvish-subtree-left)
              ("C-<left>"  . cfg-navigation/dirvish-subtree-up-to-branch)
              ("|"         . dirvish-layout-toggle)
              ("@"         . cfg-navigation/dirvish-toggle-git-msg))
  :init
  (dirvish-override-dired-mode)
  :config
  ;; =========================================================================
  ;;  Subtree & Navigation Functions (Cache Independent)
  ;; =========================================================================

  (defun cfg-navigation/dirvish-toggle-git-msg ()
    "Toggle `vc-state' and `git-msg' attributes for Dirvish."
    (interactive)
    (let ((has-git (or (memq 'git-msg dirvish-attributes)
                       (memq 'vc-state dirvish-attributes))))
      ;; Toggle the global attributes
      (if has-git
          (setq dirvish-attributes (remq 'vc-state (remq 'git-msg dirvish-attributes)))
        (setq dirvish-attributes (append dirvish-attributes '(vc-state git-msg))))

      ;; Re-initialize the active dirvish session so git background processes correctly start/stop
      (when (and (derived-mode-p 'dired-mode) (dirvish-curr))
        (let ((dir default-directory)
              (layout-p (dv-curr-layout (dirvish-curr))))
          (kill-buffer (current-buffer))
          (if layout-p
              (dirvish dir)
            (dired dir))))

      (message "Dirvish: git info %s" (if has-git "Disabled" "Enabled"))))

  (defun cfg-navigation/dirvish-last-in-subtree-p ()
    "Return t if point is on the last entry of its subtree."
    (let ((current-depth (dirvish-subtree--depth)))
      (and (> current-depth 0)
           (save-excursion
             (< (progn (forward-line 1) (dirvish-subtree--depth))
                current-depth)))))

  (defun cfg-navigation/dirvish-empty-subtree-p ()
    "Return t if on an expanded dir whose subtree has no entries."
    (and (dirvish-subtree--expanded-p)
         (save-excursion
           (forward-line 1)
           (let ((child-depth (dirvish-subtree--depth)))
             (forward-line 1)
             (< (dirvish-subtree--depth) child-depth)))))

  (defun cfg-navigation/dirvish-subtree-left ()
    "Collapse the subtree when open or move to the parent."
    (interactive)
    (if (dirvish-subtree--expanded-p)
        (dirvish-subtree-toggle)
      (dirvish-subtree-up)))

  (defun cfg-navigation/dirvish-subtree-direct-children ()
    "Return the direct children of the expanded subtree at point.
Each result is a cons of the child's buffer position and its file name."
    (let ((parent-depth (dirvish-subtree--depth))
          children)
      (save-excursion
        (forward-line 1)
        (while (and (not (eobp))
                    (> (dirvish-subtree--depth) parent-depth))
          (when (= (dirvish-subtree--depth) (1+ parent-depth))
            (push (cons (point) (dired-get-filename nil t)) children))
          (forward-line 1)))
      (nreverse children)))

  (defun cfg-navigation/dirvish-subtree-only-child-directory ()
    "Return the sole direct directory child of the subtree at point.
Return its buffer position, or nil unless the expanded subtree has
exactly one direct child and that child is a directory."
    (let ((children (cfg-navigation/dirvish-subtree-direct-children)))
      (when (and (= (length children) 1)
                 (file-directory-p (cdar children)))
        (caar children))))

  (defun cfg-navigation/dirvish-subtree-up-to-branch ()
    "Move up a single-child chain and collapse it below its first branch."
    (interactive)
    (let ((selected-item (dired-get-filename nil t)))
      (if (and selected-item (file-directory-p selected-item))
          (let ((origin (point))
                (child-position (point))
                continue
                traversed-chain
                collapsed)
            (setq continue t)
            (while continue
              (let ((origin (point)))
                (condition-case nil
                    (dirvish-subtree-up)
                  (user-error nil))
                (if (= origin (point))
                    (setq continue nil)
                  (if (> (length (cfg-navigation/dirvish-subtree-direct-children)) 1)
                      (progn
                        (goto-char child-position)
                        (when (and traversed-chain
                                   (dirvish-subtree--expanded-p))
                          (dirvish-subtree-toggle)
                          (setq collapsed t))
                        (setq continue nil))
                    (setq child-position (point)
                          traversed-chain t)))))
            (unless collapsed
              (goto-char origin)
              (cfg-navigation/dirvish-subtree-left)))
        (cfg-navigation/dirvish-subtree-left))))

  (defun cfg-navigation/dirvish-subtree-right ()
    "Expand the subtree or move to its first item when already open."
    (interactive)
    (when-let ((selected-item (dired-get-filename nil t))
               ((file-directory-p selected-item)))
      (if (dirvish-subtree--expanded-p)
          (forward-line 1)
        (dirvish-subtree-toggle))))

  (defun cfg-navigation/dirvish-subtree-expand-chain ()
    "Expand the single-directory chain rooted at the entry at point."
    (interactive)
    (when-let ((selected-item (dired-get-filename nil t))
               ((file-directory-p selected-item)))
      (let ((was-expanded (dirvish-subtree--expanded-p))
            child-position)
        (unless was-expanded
          (dirvish-subtree-toggle))
        (setq child-position (cfg-navigation/dirvish-subtree-only-child-directory))
        (if child-position
            (progn
              (while child-position
                (goto-char child-position)
                (unless (dirvish-subtree--expanded-p)
                  (dirvish-subtree-toggle))
                (setq child-position
                      (cfg-navigation/dirvish-subtree-only-child-directory)))
              (cfg-navigation/dirvish-subtree-right))
          (when was-expanded
            (cfg-navigation/dirvish-subtree-right))))))

  ;; =========================================================================
  ;;  Cache Management Functions
  ;;
  ;;  NOTE: The functions and hooks in this section are ONLY relevant if
  ;;        Dirvish cache is ENABLED. If caching is disabled (e.g. via
  ;;        `setq dirvish-reuse-copies nil`), these can remain unused.
  ;; =========================================================================

  (defun cfg-navigation/dirvish-refresh-preview-cache ()
    "Clear text preview cache + buffers; force re-read from disk.
Runs on `dirvish-after-revert-hook' so `g' reloads externally changed files."
    (when-let* ((dv (dirvish-curr)))
      (clrhash (dv-preview-hash dv))
      (clrhash (dv-parent-hash dv))
      ;; fallback reuses buffers named "PREVIEW :: TIMESTAMP :: file"; kill those
      ;; even if they were never recorded in dv-preview-buffers.
      (let ((prefix (format "PREVIEW :: %s ::" (dv-timestamp dv))))
        (dolist (buf (buffer-list))
          (when (string-prefix-p prefix (buffer-name buf))
            (dirvish--kill-buffer buf))))
      (mapc #'dirvish--kill-buffer (dv-preview-buffers dv))
      ;; Same file still selected -> debounce would skip preview update otherwise
      (dirvish--run-with-delay 'reset)
      (when-let* ((idx (dired-get-filename nil t))
                  ((dv-curr-layout dv))
                  (pwin (dv-preview-window dv))
                  ((window-live-p pwin)))
        (dirvish--preview-update dv (file-local-name idx)))))

  (defun cfg-navigation/dirvish-invalidate-saved-preview ()
    "Drop saved file from Dirvish text preview hashes so previews are not stale."
    (when buffer-file-name
      (let ((file (file-local-name (expand-file-name buffer-file-name))))
        (maphash (lambda (_id dv)
                   (remhash file (dv-preview-hash dv))
                   (remhash buffer-file-name (dv-preview-hash dv)))
                 dirvish--sessions))))

  ;; Attach Cache Invalidation Hooks
  (add-hook 'dirvish-after-revert-hook #'cfg-navigation/dirvish-refresh-preview-cache)
  (add-hook 'after-save-hook #'cfg-navigation/dirvish-invalidate-saved-preview)

  ;; =========================================================================
  ;;  General Settings & Options
  ;; =========================================================================

  (setq dirvish-reuse-copies nil) ; Disable cached preview buffers when navigating -- solves issue with stale files
  (setq dirvish-mode-line-format '(:left (path) :right (file-size)))
  (setq dirvish-attributes '(subtree-state nerd-icons file-size))
  (setq dirvish-preview-dispatchers '(dired image gif audio epub archive font pdf video))
  (setq dirvish-cache-dir
        (expand-file-name "dirvish/" user-emacs-directory))
  ;; "-l" is required for showing icons in subtrees. See note on `dired-listing-switches`:
  (setq dirvish-subtree-listing-switches "-Ahl --group-directories-first")

  ;; Configuring Auto Revert mode for Dirvish/Dired buffers
  (setq global-auto-revert-non-file-buffers t) ; Allow Auto Revert monitor Dired/Dirvish buffers
  (setq auto-revert-verbose nil) ; Stop Emacs from printing "Reverting buffer..." in the echo area every time it updates
  (add-hook 'dired-mode-hook #'auto-revert-mode))

(defun cfg-navigation/rg-search-in-subtree ()
  "Prompt for a search string and run rg in the folder at point."
  (interactive)
  (require 'rg) ; Ensure the package and its variables are loaded
  (defvar rg-custom-flags)
  (let ((target-dir (dired-get-filename nil t))
        (flags (if (boundp 'rg-custom-flags)
                   (cons "--hidden" rg-custom-flags)
                 '("--hidden"))))

    (unless (and target-dir (file-directory-p target-dir))
      (setq target-dir (file-name-directory (dired-get-filename))))

    (let ((default-directory target-dir)
          (rg-custom-flags flags))
      (call-interactively #'rg))))

install__tools__dired__dirvish_utilities_bundle() {
  log_info "Starting Dirvish utilities setup..."

  if [ -f /etc/debian_version ]; then
    # Debian/Ubuntu: There is no official Personal Package Archive (PPA) for Glow on Ubuntu.
    # Instead, the developers provide a dedicated APT repository and other official installation methods.
    sudo mkdir -p /etc/apt/keyrings
    if [ ! -f /etc/apt/keyrings/charm.gpg ]; then
      curl -fsSL https://repo.charm.sh/apt/gpg.key | sudo gpg --dearmor -o /etc/apt/keyrings/charm.gpg
    fi
    echo "deb [signed-by=/etc/apt/keyrings/charm.gpg] https://repo.charm.sh/apt/ * *" | sudo tee /etc/apt/sources.list.d/charm.list
  fi

  install_system_packages "${DIRVISH_UTILITIES[@]}"
