* Description
Combines the Spacemacs-style leader-key philosophy with a priority minor mode
(~cfg-keys-mode~) to manage global shortcuts consistently. Provides core keyboard
navigation enhancements using avy for fast visual jumping, goto-line-preview for
interactive line jumping, and mwim for smart home/end line position movement.

* Emacs Configuration
** Minor Mode & Leader Map Setup
We define a high-priority global minor mode (~cfg-keys-mode~) and bind ~C-c~ to
act as the primary leader key prefix via ~cfg-keys-leader-map~.

(defvar cfg-keys-map (make-sparse-keymap)
  "Keymap used for user custom bindings.")

(define-minor-mode cfg-keys-mode
  "A minor mode for custom keybindings that override major modes."
  :global t
  :group 'keybindings
  :keymap cfg-keys-map)

(cfg-keys-mode 1)
(defvar cfg-keys-leader-map)
(define-prefix-command 'cfg-keys-leader-map)
(define-key cfg-keys-map (kbd "C-c") 'cfg-keys-leader-map)

** Leader Definition Macro
The ~cfg-keys/leader-def~ macro simplifies registering pairs of key sequences
and interactive functions directly under the leader keymap. Each binding case
delegates to its own small helper function, so keymap-conflict detection
lives in one place per case instead of inline in the macro expansion.

Conflicts are detected dynamically:
1. Exact same prefix label re-declaration: skipped silently.
2. Conflicting prefix label re-declaration: logged to ~*Messages*~, keymap kept.
3. Overwriting a non-prefix shortcut: logged to ~*Messages*~, new binding skipped.

#+begin_src emacs-lisp
(defun cfg-keys--log-conflict (key existing target)
  "Log to `*Messages*' that KEY is already bound to EXISTING, skipping TARGET."
  (message "[cfg-keys] Conflict detected! Key sequence \"%s\" already bound to `%s`. Skipping registration of `%s`."
           key existing target))

(defun cfg-keys--raw-binding (map key)
  "Return the raw (EVENT LABEL? . DEF) alist cell for KEY's final event in MAP.
`lookup-key' strips the LABEL string off a `(LABEL . KEYMAP)' prefix binding
before returning it, so label comparisons must read the keymap alist
directly instead."
  (let* ((keyseq (kbd key))
         (n (length keyseq)))
    (when (> n 0)
      (let* ((last-event (aref keyseq (1- n)))
             (parent (if (> n 1) (lookup-key map (substring keyseq 0 (1- n))) map)))
        (and (keymapp parent) (assq last-event (cdr parent)))))))

(defun cfg-keys--bind-prefix-label (key label)
  "Bind KEY as a prefix keymap labeled LABEL under `cfg-keys-leader-map'.
Reuses an existing keymap at KEY when present. Logs to `*Messages*' when
LABEL differs from a prefix label already installed at KEY."
  (let* ((keyseq (kbd key))
         (raw-entry (cfg-keys--raw-binding cfg-keys-leader-map key))
         (existing-label (and raw-entry (stringp (cadr raw-entry)) (cadr raw-entry)))
         (existing-target (cond (existing-label (cddr raw-entry))
                                 (raw-entry (cdr raw-entry))
                                 (t nil)))
         (km (if (keymapp existing-target) existing-target (make-sparse-keymap))))
    (cond
     ((and (keymapp existing-target) (equal existing-label label))
      nil)
     ((and (keymapp existing-target) existing-label (not (equal existing-label label)))
      (message "[cfg-keys] Prefix label conflict for key \"%s\": changing \"%s\" -> \"%s\""
                key existing-label label)
      (define-key cfg-keys-leader-map keyseq (cons label km)))
     (t
      (define-key cfg-keys-leader-map keyseq (cons label km))))))

(defun cfg-keys--bind-labeled-command (key label cmd)
  "Bind CMD to KEY under `cfg-keys-leader-map' with which-key LABEL.
Skips the binding when KEY already carries a conflicting non-prefix binding."
  (let* ((keyseq (kbd key))
         (existing (lookup-key cfg-keys-leader-map keyseq)))
    (if (and existing (not (numberp existing)))
        (cfg-keys--log-conflict key existing cmd)
      (define-key cfg-keys-leader-map keyseq (cons label cmd)))))

(defun cfg-keys--bind-plain (key def)
  "Bind DEF to KEY under `cfg-keys-leader-map'.
Skips the binding when KEY already carries a conflicting non-prefix binding."
  (let* ((keyseq (kbd key))
         (existing (lookup-key cfg-keys-leader-map keyseq)))
    (if (and existing (not (numberp existing)))
        (cfg-keys--log-conflict key existing def)
      (define-key cfg-keys-leader-map keyseq def))))

(defmacro cfg-keys/leader-def (&rest keybinds)
  "Convenience macro to bind KEYBINDS under `cfg-keys-leader-map`.
Usage:
`(cfg-keys/leader-def \"f f\" #\\='find-file \"b b\" #\\='switch-to-buffer)`.
Prefix labels use `(:ignore t :which-key \"name\")`. Those install a labeled
sparse keymap (or label an existing one) instead of a non-prefix binding, so
nested sequences like \"t f\" remain valid.
Commands may also carry a which-key description via `(command :which-key
\"desc\")`, which binds COMMAND with DESC as its which-key label.

Detects conflicts dynamically via `cfg-keys--bind-prefix-label',
`cfg-keys--bind-labeled-command', and `cfg-keys--bind-plain':
1. Exact same prefix label re-declaration: skipped silently.
2. Conflicting prefix label re-declaration: logged to `*Messages*'.
3. Overwriting non-ignore shortcut: logged to `*Messages*' and skipped."
  (declare (indent 0))
  `(progn
     ,@(cl-loop
        for (key fn) on keybinds by #'cddr
        for form = (if (eq (car-safe fn) 'quote) (cadr fn) fn)
        for ignore-label = (and (eq (car-safe form) :ignore)
                                (plist-get form :which-key))
        for cmd-label = (and (consp form)
                              (not (eq (car form) :ignore))
                              (not (eq (car form) 'function))
                              (not (eq (car form) 'lambda))
                              (plist-get (cdr form) :which-key))
        collect
        (cond
         (ignore-label
          `(cfg-keys--bind-prefix-label ,key ,ignore-label))
         (cmd-label
          `(cfg-keys--bind-labeled-command ,key ,cmd-label #',(car form)))
         (t
          `(cfg-keys--bind-plain ,key ,fn))))))

** Mode-local Leader Definition
Define mode-specific leader bindings using the same syntax and interface as
~cfg-keys/leader-def~.

Registers bindings to a buffer-local leader keymap constructed on ~MODE-hook~.
This mode-specific map automatically inherits from the global leader keymap,
mirroring its global prefix structures.

- *Global Access*: Keeps global leader bindings accessible alongside
  mode-specific ones.
- *Robust Mapping*: Rebuilds fresh maps on hook execution to prevent stale
  parent references or cyclic inheritance.
- *Override Protection*: Emits a warning whenever a mode-specific binding
  overrides a global command.

#+begin_src emacs-lisp
(defvar cfg-keys--mode-bindings (make-hash-table :test #'eq)
  "MODE -> list of (KEY . SPEC) for `cfg-keys/mode-leader-def'.
SPEC is (:prefix LABEL), (LABEL . COMMAND), or COMMAND.")

(defvar cfg-keys--mode-hooks-installed (make-hash-table :test #'eq)
  "Modes that already have a mode-leader install hook.")

(defun cfg-keys--unwrap-keymap (def)
  "Return keymap from DEF, unwrapping which-key `(LABEL . KEYMAP)' conses."
  (cond ((and (listp def) (keymapp def)) def)
        ((and (consp def) (stringp (car def))
              (listp (cdr def)) (keymapp (cdr def)))
         (cdr def))))

(defun cfg-keys--map-own (map event)
  "Return MAP's own binding for EVENT (ignore parent)."
  (let ((raw (assq event (cdr map))))
    (and raw
         (if (and (consp (cdr raw)) (stringp (cadr raw)))
             (cddr raw)
           (cdr raw)))))

(defun cfg-keys--mode-bind (leader key def)
  "Bind KEY to DEF on LEADER without mutating parent keymaps.
Creates local prefix maps parented to the matching global prefixes."
  (let* ((keyseq (kbd key))
         (n (length keyseq))
         (map leader)
         (prefix []))
    (dotimes (i n)
      (let* ((event (aref keyseq i))
             (last (= i (1- n)))
             (own (cfg-keys--unwrap-keymap (cfg-keys--map-own map event))))
        (setq prefix (vconcat prefix (vector event)))
        (cond
         (last (define-key map (vector event) def))
         ((and (listp own) (keymapp own)) (setq map own))
         (t
          (let* ((g (cfg-keys--unwrap-keymap
                     (lookup-key cfg-keys-leader-map prefix)))
                 (raw (cfg-keys--raw-binding
                       cfg-keys-leader-map (key-description prefix)))
                 (label (and raw (stringp (cadr raw)) (cadr raw)))
                 (km (make-sparse-keymap)))
            (when g (set-keymap-parent km g))
            (define-key map (vector event) (if label (cons label km) km))
            (setq map km))))))))

(defun cfg-keys--normalize-modes (modes)
  "Return MODES as a list of major-mode symbols."
  (cond
   ((symbolp modes) (list modes))
   ((and (listp modes) (cl-every #'symbolp modes)) modes)
   (t (error "[cfg-keys] mode-leader-def expected a mode or list of modes, got %S"
             modes))))

(defun cfg-keys--mode-register (mode key spec)
  "Record SPEC for KEY under MODE.
Identical re-registration is a no-op. Different SPEC on the same key logs a
conflict. Keys on other modes are independent (no cross-mode conflict)."
  (let* ((entries (gethash mode cfg-keys--mode-bindings))
         (existing (alist-get key entries nil nil #'equal)))
    (cond
     ((and existing (equal existing spec)) nil)
     (existing (cfg-keys--log-conflict key existing spec))
     (t (puthash mode (cons (cons key spec) entries)
                 cfg-keys--mode-bindings)))))

(defun cfg-keys--mode-apply (leader mode key spec)
  "Apply SPEC for KEY onto LEADER under MODE.
Warn only when shadowing a different command."
  (let* ((g (lookup-key cfg-keys-leader-map (kbd key)))
         (gcmd (and g (not (numberp g)) (not (cfg-keys--unwrap-keymap g)) g)))
    (pcase spec
      (`(:prefix ,label)
       (let* ((existing (cfg-keys--unwrap-keymap (lookup-key leader (kbd key))))
              (km (or existing (make-sparse-keymap)))
              (gkm (cfg-keys--unwrap-keymap g)))
         (unless existing
           (when gkm (set-keymap-parent km gkm)))
         (cfg-keys--mode-bind leader key (cons label km))))
      ((and `(,label . ,cmd) (guard (stringp label)))
       (when (and gcmd (not (eq gcmd cmd)))
         (message "[cfg-keys] Mode `%s' overrides global \"%s\" (`%s' -> `%s')."
                  mode key gcmd cmd))
       (cfg-keys--mode-bind leader key (cons label cmd)))
      (cmd
       (when (and gcmd (not (eq gcmd cmd)))
         (message "[cfg-keys] Mode `%s' overrides global \"%s\" (`%s' -> `%s')."
                  mode key gcmd cmd))
       (cfg-keys--mode-bind leader key cmd)))))

(defun cfg-keys--leader-keyseqs ()
  "Return key sequences on `cfg-keys-map' bound to `cfg-keys-leader-map'."
  (let (keys)
    (map-keymap-internal
     (lambda (event def)
       (when (eq def 'cfg-keys-leader-map)
         (push (vector event) keys)))
     cfg-keys-map)
    keys))

(defun cfg-keys--install-mode-leader (mode)
  "Activate MODE's leader bindings in the current buffer."
  (let* ((leader (make-sparse-keymap))
         (map (make-sparse-keymap))
         (leader-keys (cfg-keys--leader-keyseqs)))
    (unless leader-keys
      (error "[cfg-keys] No leader key bound to `cfg-keys-leader-map' on `cfg-keys-map'"))
    (set-keymap-parent leader cfg-keys-leader-map)
    (dolist (entry (reverse (gethash mode cfg-keys--mode-bindings)))
      (cfg-keys--mode-apply leader mode (car entry) (cdr entry)))
    (set-keymap-parent map cfg-keys-map)
    (dolist (key leader-keys)
      (define-key map key leader))
    (setq-local minor-mode-overriding-map-alist
                (cons (cons 'cfg-keys-mode map)
                      (assq-delete-all
                       'cfg-keys-mode
                       (copy-sequence
                        (or minor-mode-overriding-map-alist
                            (default-value 'minor-mode-overriding-map-alist)
                            '())))))))

(defun cfg-keys--ensure-mode-hook (mode)
  "Add a one-shot `MODE-hook' for MODE to install mode-local leader bindings."
  (unless (gethash mode cfg-keys--mode-hooks-installed)
    (add-hook (intern (format "%s-hook" mode))
              (lambda () (cfg-keys--install-mode-leader mode)))
    (puthash mode t cfg-keys--mode-hooks-installed)))

(defmacro cfg-keys/mode-leader-def (modes &rest keybinds)
  "Like `cfg-keys/leader-def', but KEYBINDS apply only in MODES buffers.
MODES is a major-mode symbol or a list of symbols."
  (declare (indent 1))
  `(let ((modes (cfg-keys--normalize-modes ',modes)))
     (dolist (mode modes)
       (cfg-keys--ensure-mode-hook mode))
     ,@(cl-loop
        for (key fn) on keybinds by #'cddr
        for form = (if (eq (car-safe fn) 'quote) (cadr fn) fn)
        for ignore-label = (and (eq (car-safe form) :ignore)
                                (plist-get form :which-key))
        for cmd-label = (and (consp form)
                              (not (eq (car form) :ignore))
                              (not (eq (car form) 'function))
                              (not (eq (car form) 'lambda))
                              (plist-get (cdr form) :which-key))
        collect
        (cond
         (ignore-label
          `(dolist (mode modes)
             (cfg-keys--mode-register mode ,key (list :prefix ,ignore-label))))
         (cmd-label
          `(dolist (mode modes)
             (cfg-keys--mode-register mode ,key
                                      (cons ,cmd-label #',(car form)))))
         (t
          `(dolist (mode modes)
             (cfg-keys--mode-register mode ,key ,fn)))))))
#+end_src

** Keybindings Tree Printer
Renders all bindings under ~cfg-keys-leader-map~ as a tree in a dedicated
buffer, so the full ~C-c~ leader map can be inspected at a glance instead of
navigating ~which-key~ one level at a time.

Always appends a second section for mode-local leader keys registered via
~cfg-keys/mode-leader-def~ (from ~cfg-keys--mode-bindings~). Those are listed
per major mode, using the same tree layout as the global map, without inheriting
global prefix contents.

#+begin_src emacs-lisp
(defun cfg-keys--unwrap-def (def)
  "Extract (REAL-DEF . LABEL) from DEF.
Handles `which-key` cons-cell pairs generated by `cfg-keys/leader-def`."
  (if (and (consp def) (stringp (car def)))
      (cons (cdr def) (car def))
    (cons def nil)))

(defun cfg-keys--format-target-string (real-def label)
  "Format target description string using REAL-DEF and optional LABEL."
  (let ((cmd-name (cond
                   ((compiled-function-p real-def) "<compiled-lambda>")
                   ((and (consp real-def) (eq (car real-def) 'closure)) "<closure>")
                   ((and (consp real-def) (eq (car real-def) 'lambda)) "<lambda>")
                   ((symbolp real-def) (symbol-name real-def))
                   (t (format "%s" real-def)))))
    (if label
        (format "%s (%s)" cmd-name label)
      cmd-name)))

(defun cfg-keys--format-tree-line (key-str target depth is-last)
  "Format a tree line for KEY-STR and TARGET at DEPTH.
If IS-LAST is non-nil, use a corner leaf connector."
  (let* ((connector (if is-last "└── " "├── "))
         (indent (if (> depth 0)
                     (concat (make-string (* (1- depth) 4) ?\s) connector)
                   connector)))
    (format "  %s%-10s ->  %s\n" indent key-str target)))

(defun cfg-keys--render-prefix-tree (km full-prefix depth)
  "Recursively traverse keymap KM under FULL-PREFIX at DEPTH and return string."
  (let ((output "")
        (entries '()))
    (when (keymapp km)
      (map-keymap
       (lambda (event def)
         (let* ((key-str (single-key-description event))
                (unwrapped (cfg-keys--unwrap-def def)))
           (push (list key-str (car unwrapped) (cdr unwrapped)) entries)))
       km)

      (setq entries (sort entries (lambda (a b) (string< (car a) (car b)))))

      (let ((total (length entries))
            (idx 0))
        (dolist (entry entries)
          (setq idx (1+ idx))
          (let* ((key-str (nth 0 entry))
                 (real-def (nth 1 entry))
                 (label (nth 2 entry))
                 (is-last (= idx total))
                 (next-prefix (concat full-prefix " " key-str)))
            (cond
             ((keymapp real-def)
              (setq output
                    (concat output
                            (format "  %s%s [%s] (C-c %s)\n"
                                    (if (> depth 0) (make-string (* depth 4) ?\s) "")
                                    (if is-last "└──" "├──")
                                    (or label "Prefix")
                                    next-prefix)
                            (cfg-keys--render-prefix-tree real-def next-prefix (1+ depth)))))
             (t
              (let ((target (cfg-keys--format-target-string real-def label)))
                (setq output
                      (concat output
                              (cfg-keys--format-tree-line key-str target depth is-last))))))))))
    output))

(defun cfg-keys--leader-map-parts (leader-map)
  "Return (DIRECT-BINDINGS . PREFIX-GROUPS) from LEADER-MAP.
DIRECT-BINDINGS is a sorted alist of (KEY-STRING . TARGET-STRING).
PREFIX-GROUPS is a sorted list of (KEY-STRING KEYMAP LABEL)."
  (let ((direct-bindings '())
        (prefix-groups '()))
    (when (keymapp leader-map)
      (map-keymap
       (lambda (event def)
         (let* ((key-str (single-key-description event))
                (unwrapped (cfg-keys--unwrap-def def))
                (real-def (car unwrapped))
                (label (cdr unwrapped)))
           (if (keymapp real-def)
               (push (list key-str real-def label) prefix-groups)
             (push (cons key-str (cfg-keys--format-target-string real-def label))
                   direct-bindings))))
       leader-map))
    (cons (sort direct-bindings (lambda (a b) (string< (car a) (car b))))
          (sort prefix-groups (lambda (a b) (string< (car a) (car b)))))))

(defun cfg-keys--insert-leader-parts (direct-bindings prefix-groups)
  "Insert DIRECT-BINDINGS then PREFIX-GROUPS into the current buffer."
  (when direct-bindings
    (insert "Direct Leader Shortcuts:\n")
    (insert (make-string 30 ?-) "\n")
    (dolist (b direct-bindings)
      (insert (format "  C-c %-10s ->  %s\n" (car b) (cdr b))))
    (insert "\n"))
  (when prefix-groups
    (insert "Prefix Groups:\n")
    (insert (make-string 30 ?-) "\n")
    (dolist (group prefix-groups)
      (let* ((key-str (nth 0 group))
             (km (nth 1 group))
             (label (nth 2 group))
             (full-prefix (concat "C-c " key-str)))
        (insert (format "\n[%s] (%s)\n" (or label "Prefix") full-prefix))
        (insert (make-string 45 ?-) "\n")
        (insert (cfg-keys--render-prefix-tree km key-str 0))))))

(defun cfg-keys--mode-leader-keymap (mode)
  "Build a keymap of MODE's `cfg-keys/mode-leader-def' bindings only.
Does not parent prefix maps to the global leader, so the printer does not
mix in global keys."
  (let ((leader (make-sparse-keymap)))
    (dolist (entry (reverse (gethash mode cfg-keys--mode-bindings)))
      (let ((key (car entry))
            (spec (cdr entry)))
        (pcase spec
          (`(:prefix ,label)
           (let* ((existing (lookup-key leader (kbd key)))
                  (km (or (cfg-keys--unwrap-keymap existing)
                          (make-sparse-keymap))))
             (define-key leader (kbd key) (cons label km))))
          ((and `(,label . ,cmd) (guard (stringp label)))
           (define-key leader (kbd key) (cons label cmd)))
          (cmd
           (define-key leader (kbd key) cmd)))))
    leader))

(defun cfg-keys--sorted-mode-leader-modes ()
  "Return major modes that have `cfg-keys/mode-leader-def' bindings, sorted."
  (let (modes)
    (maphash (lambda (mode _bindings) (push mode modes))
             cfg-keys--mode-bindings)
    (sort modes (lambda (a b) (string< (symbol-name a) (symbol-name b))))))

(defun cfg-keys--insert-mode-leader-section ()
  "Insert per-mode leader trees from `cfg-keys--mode-bindings'."
  (insert "\n\nMode-specific Leader Bindings (`cfg-keys/mode-leader-def`):\n")
  (insert (make-string 65 ?=) "\n")
  (let ((modes (cfg-keys--sorted-mode-leader-modes)))
    (if (null modes)
        (insert "\n  No mode-specific leader bindings registered.\n")
      (dolist (mode modes)
        (let* ((name (symbol-name mode))
               (parts (cfg-keys--leader-map-parts
                       (cfg-keys--mode-leader-keymap mode)))
               (direct (car parts))
               (prefixes (cdr parts)))
          (insert (format "\n=%s=\n" name))
          (insert (make-string (max 30 (length name)) ?=) "\n\n")
          (if (and (null direct) (null prefixes))
              (insert "  No bindings.\n")
            (cfg-keys--insert-leader-parts direct prefixes)))))))

(defun cfg-keys--collect-non-leader-bindings ()
  "Collect bindings in `cfg-keys-map` other than the leader prefix.
Returns a sorted alist of (KEY-STRING . TARGET-STRING)."
  (let ((bindings '()))
    (when (boundp 'cfg-keys-map)
      (map-keymap
       (lambda (event def)
         (unless (eq def 'cfg-keys-leader-map)
           (let* ((key-str (single-key-description event))
                  (unwrapped (cfg-keys--unwrap-def def))
                  (real-def (car unwrapped))
                  (label (cdr unwrapped)))
             (push (cons key-str (cfg-keys--format-target-string real-def label)) bindings))))
       cfg-keys-map))
    (sort bindings (lambda (a b) (string< (car a) (car b))))))

(defun cfg-keys/print-tree (&optional show-all)
  "Print custom leader keybindings (`cfg-keys-leader-map') formatted as a tree.
Always includes mode-local leader keys from `cfg-keys/mode-leader-def'.
With prefix argument SHOW-ALL (\\[universal-argument]), also prints non-leader
bindings in `cfg-keys-map` at the bottom."
  (interactive "P")
  (let ((buf (get-buffer-create "*CFG Keybindings Tree*"))
        (leader-map (and (boundp 'cfg-keys-leader-map)
                         (symbol-value 'cfg-keys-leader-map)))
        (other-bindings (when show-all (cfg-keys--collect-non-leader-bindings))))

    (unless (keymapp leader-map)
      (error "`cfg-keys-leader-map` is not defined or valid"))

    (with-current-buffer buf
      (read-only-mode -1)
      (erase-buffer)

      (insert "Custom Leader Keybindings Tree (`C-c` Leader Map)\n")
      (insert (make-string 65 ?=) "\n\n")

      (let ((parts (cfg-keys--leader-map-parts leader-map)))
        (cfg-keys--insert-leader-parts (car parts) (cdr parts)))

      (cfg-keys--insert-mode-leader-section)

      (when show-all
        (insert "\n\nOther Mode Map Bindings (`cfg-keys-map`):\n")
        (insert (make-string 65 ?=) "\n\n")
        (if (null other-bindings)
            (insert "  No additional non-leader bindings found.\n")
          (dolist (b other-bindings)
            (insert (format "  %-12s ->  %s\n" (car b) (cdr b))))))

      (special-mode)
      (display-buffer buf))))

(defun cfg-keys/print-tree-all ()
  "Print global+mode-local leader keys, and non-leader `cfg-keys-map' bindings."
  (interactive)
  (cfg-keys/print-tree t))

** MWIM (Move Where I Mean)
MWIM (Move Where I Mean) is a popular Emacs package that makes your "Home" and
"End" keys (or =C-a=/=HOME= and =C-e=/=END=) smarter.

Standard Emacs commands are literal: =C-a=/=HOME= goes to the very beginning of
the line (column 0), and =C-e=/=END= goes to the very end. MWIM changes this to
behave more like modern IDEs.

#+begin_src emacs-lisp
(global-set-key [remap goto-line] 'goto-line-preview)
