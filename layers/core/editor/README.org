#+TITLE: Core Editor Layer
#+CATEGORY: core
#+PROPERTY: header-args:emacs-lisp :tangle yes

* Description
Configures fundamental editing defaults, text manipulation rules, and buffer
behaviors across programming and prose modes. This layer establishes baseline
encoding and line ending standards, centralizes backup and auto-save file
management, sets up unified indentation defaults, handles EditorConfig rules,
ANSI color interpretation, and line wrapping alongside visual editing aids like line numbers and whitespace indicators.

* Emacs Configuration
** ANSI Color Support
Translate ANSI color codes into text properties in compilation buffers.

#+begin_src emacs-lisp
(use-package ansi-color
  :ensure nil ; Built-in
  :config
  (defun cfg-core/ansi-colorize-buffer ()
    "Translate ANSI color codes into text properties."
    (let ((inhibit-read-only t))
      (read-only-mode -1)
      (ansi-color-apply-on-region (point-min) (point-max))
      (read-only-mode 1)))

  (add-hook 'compilation-filter-hook 'ansi-color-compilation-filter))
#+end_src

** Backups & Auto-Saves
Redirect backup files and auto-save files away from working project directories
into dedicated subdirectories within the Emacs configuration directory.

#+begin_src emacs-lisp
(setq create-lockfiles nil)

(setq backup-directory-alist
      `((".*" . ,(expand-file-name "backups/" user-emacs-directory))))

(make-directory (expand-file-name "auto-save-list/" user-emacs-directory) t)

      `((".*" ,(expand-file-name "auto-save-list/" user-emacs-directory) t)))
#+end_src

** EditorConfig Integration
Using [[https://editorconfig.org/][EditorConfig]] is a must when collaborating with others. It is also a way
of having multiple tools that want to format your buffer to agree (e.g. both
the language's Emacs mode and some external formatter/prettifier).

#+begin_src emacs-lisp
(use-package editorconfig
  :config
  (add-to-list 'editorconfig-indentation-alist '(json-mode json-indent-level))
  (add-to-list 'editorconfig-indentation-alist '(json-ts-mode json-ts-mode-indent-level))
  (add-to-list 'editorconfig-indentation-alist '(js-json-mode js-indent-level))
  (editorconfig-mode 1))
#+end_src

** Save Hooks
Run a configurable list of functions before saving. Currently trims trailing
whitespace in selected modes.

#+begin_src emacs-lisp
(defun cfg-core/trim-whitespace ()
  "Trim trailing whitespace before saving."
  (when (derived-mode-p
    'conf-unix-mode
    'markdown-mode
    'nxml-mode
    'org-mode
    'prog-mode
    'text-mode
    'yaml-ts-mode) ; only certain modes
    (delete-trailing-whitespace)))

(defvar cfg-core/save-hooks
  '(cfg-core/trim-whitespace
    ;; more functions here
    )
  "List of functions to run before saving a file.")

(defun cfg-core/run-save-hooks ()
  "Run all functions in `cfg-core/save-hooks'."
  (run-hooks 'cfg-core/save-hooks))

(add-hook 'before-save-hook #'cfg-core/run-save-hooks)
#+end_src

** Indentation Framework
Defines a standard 2-space indentation application function across text and
programming hooks. Special care is taken to avoid interfering with Org mode tab
widths and to set tree-sitter defaults ahead of mode initialization.

#+begin_src emacs-lisp
(defvar sh-basic-offset)
(defvar sh-indentation)
(defvar c-basic-offset)
(defvar java-ts-mode-indent-offset)
(defvar js-indent-level)
(defvar typescript-indent-level)
(defvar css-indent-offset)
(defvar python-indent-offset)
(defvar nxml-child-indent)
(defvar coffee-tab-width)

(setq-default java-ts-mode-indent-offset 2)

(defun cfg-core/default-indentation ()
  "Apply preferred 2-space indentation settings across modes.
These settings are intended to be overridden by `editorconfig-mode'."
  (unless (eq major-mode 'org-mode)
    (setq
     tab-width 2
     indent-tabs-mode nil
     standard-indent 2
     sh-basic-offset 2
     sh-indentation 2
     c-basic-offset 2
     java-ts-mode-indent-offset 2
     js-indent-level 2
     typescript-indent-level 2
     css-indent-offset 2
     python-indent-offset 2
     nxml-child-indent 2
     coffee-tab-width 2)))

(dolist (hook '(conf-unix-mode-hook
                java-ts-mode-hook
                markdown-mode-hook
                nxml-mode-hook
                prog-mode-hook
                text-mode-hook
                yaml-ts-mode-hook))
  (add-hook hook #'cfg-core/default-indentation))
#+end_src

** General Whitespace and Special Characters Visualization
Highlight tabs + non-ASCII spaces only. Normal ASCII spaces and newlines stay
plain. Trailing whitespace uses a wave underline. Unicode homoglyphs + fancy
spaces use a high-contrast warning face.

#+begin_src emacs-lisp
(defface cfg-core/homoglyph-face
  '((t :foreground "yellow"
       :background "darkred"
       :weight bold
       :box (:line-width (3 . -1) :color "darkred")
       :underline (:style wave :color "yellow")))
  "High-visibility face for Unicode homoglyphs and non-ASCII spaces.")

(use-package whitespace
  :ensure nil ; built-in
  :config
  (global-whitespace-mode 1))

(declare-function whitespace-display-char-off "whitespace")
(declare-function whitespace-display-char-on "whitespace")
(use-package unicode-whitespace
  :after whitespace
  :config
  (unicode-whitespace-setup 'subdued-faces)
  ;; unicode-whitespace-setup overwrites whitespace-style + maps ASCII Space ->
  ;; Middle Dot, and re-enables newline-mark when truncate-lines/word-wrap is
  ;; on (even if newline-mark-modes is empty). Reassert: tabs + non-ASCII
  ;; spaces only; no normal spaces; no newlines.
  (setq unicode-whitespace-newline-mark-modes nil)
  (setq whitespace-style
        '(face
          tabs
          tab-mark
          space-before-tab
          space-after-tab
          spaces
          space-mark))
  ;; Drop ordinary Space (U+0020) from soft-space recognition + glyph mapping.
  (setq whitespace-space-regexp
        (replace-regexp-in-string " " "" whitespace-space-regexp))
  (setq whitespace-display-mappings
        (seq-filter (lambda (m)
                      (and (not (eq (car m) 'newline-mark))
                           (not (eq (nth 1 m) ?\s))))
                    whitespace-display-mappings))
  ;; Restore tab arrow (unicode-whitespace skips ordinary TAB mappings).
  (add-to-list 'whitespace-display-mappings
               '(tab-mark ?\t [?\u2192 ?\t] [?\t]))
  ;; whitespace applies space faces with override `t`, so point them at
  ;; cfg-core/homoglyph-face -- troll-stopper prepend cannot win on spaces.
  (dolist (face '(unicode-whitespace-subdued-space
                  unicode-whitespace-subdued-hspace
                  whitespace-space
                  whitespace-hspace))
    (when (facep face)
      (set-face-attribute face nil :inherit 'cfg-core/homoglyph-face
                          :foreground 'unspecified
                          :background 'unspecified
                          :weight 'unspecified
                          :underline 'unspecified)))
  ;; Kill newline marks if unicode-whitespace hook re-adds newline-mark.
  (add-hook 'whitespace-mode-hook
            (lambda ()
              (when whitespace-mode
                (setq-local whitespace-active-style
                            (remq 'newline-mark
                                  (remq 'newline
                                        (copy-sequence
                                         (or whitespace-active-style
                                             whitespace-style)))))
                (whitespace-display-char-off)
                (whitespace-display-char-on)))
            100)
  (global-whitespace-mode 0)
  (global-whitespace-mode 1))

(dolist (hook '(prog-mode-hook text-mode-hook))
  (add-hook hook
            (lambda ()
              (setq-local show-trailing-whitespace t)
              (face-remap-add-relative
               'trailing-whitespace
               `(:background ,(face-background 'default)
                             :underline (:style wave :color ,(face-foreground 'error)))))))

(use-package unicode-troll-stopper
  :after unicode-whitespace
  :hook ((prog-mode . unicode-troll-stopper-mode)
         (text-mode . unicode-troll-stopper-mode))
  :config
  ;; Package bug: (regexp-opt STRINGS 'words) wraps the pattern in \<...\>. That
  ;; only matches word-syntax chars. Leading Unicode spaces (NBSP, EN/EM
  ;; QUAD/SPACE, ...; syntax class " ") and several others never match. Strip
  ;; the  word-boundary anchors. Use override `t` so this wins over other
  ;; font-lock faces on non-spaces.
  (let ((re (caar unicode-troll-stopper--keywords)))
    (when (string-prefix-p "\\<" re)
      (setq re (substring re 2)))
    (when (string-suffix-p "\\>" re)
      (setq re (substring re 0 -2)))
    (setq unicode-troll-stopper--keywords
          `((,re 1 'cfg-core/homoglyph-face t)))))
#+end_src

** Modes configuration
Enable essential built-in global modes for buffer editing and file management.

#+begin_src emacs-lisp
(dolist (mode
         '(abbrev-mode             ; E.g. sopl -> System.out.println
           delete-selection-mode   ; Replace selected text
           dirtrack-mode           ; directory tracking in *shell*
           global-auto-revert-mode ; Revert files when changed on disk
           global-so-long-mode     ; Mitigate performance for long lines
           recentf-mode))          ; Recently opened files
  (funcall mode 1))
#+end_src

leader toggle for the fill-column indicator.

(cfg-keys/leader-def
  "t"   '(:ignore t :which-key "toggles")
