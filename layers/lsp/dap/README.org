#+TITLE: DAP Layer
#+CATEGORY: lsp
#+PROPERTY: header-args:emacs-lisp :tangle yes

* Description
Provides debugging support via the Debug Adapter Protocol using dap-mode. Includes
smart F-key dispatch, force-quit utilities, a custom Stop button, fringe-click
breakpoints, and UI freeze protection for large debug sessions.

** DAP Mode

(defun cfg-lsp/dap--thread-stopped-p (session)
  "Return non-nil if SESSION has a currently stopped thread."
  (and session
       (dap--session-running session)
       (dap--debug-session-thread-id session)))

(defun cfg-lsp/dap-run-continue-dispatch ()
  "Smart F5: continue a stopped thread, or start a new debug session.
- If a thread is stopped (breakpoint/exception): calls `dap-continue'.
- If session is running but no thread is stopped: shows a hint.
- Otherwise: starts a new session via `cfg-lsp/dap-start-debug'."
  (interactive)
  (let ((session (and (fboundp 'dap--cur-session) (dap--cur-session))))
    (cond
     ((cfg-lsp/dap--thread-stopped-p session)
      (call-interactively 'dap-continue))
     ((and session (dap--session-running session))
      (message "Debug session is running. Set a breakpoint (<f9>) and trigger it to pause."))
     (t
      (if (funcall cfg-lsp/dap-check-requirements)
          (funcall cfg-lsp/dap-start-debug)
        (message "DAP pre-flight check failed; not starting debug session."))))))

(defun cfg-lsp/dap-step-dispatch (step-fn)
  "Call STEP-FN if a thread is stopped, otherwise show a hint."
  (let ((session (and (fboundp 'dap--cur-session) (dap--cur-session))))
    (if (cfg-lsp/dap--thread-stopped-p session)
        (call-interactively step-fn)
      (message "No stopped thread. Set a breakpoint (<f9>) and trigger it first."))))

  ;; command, which returns ugly JSON and can't handle simple math like "a + 1".

** DAP Force Quit and Extended Controls
The default DAP controls posframe has no dedicated "Force Stop" button --
~dap-disconnect~ uses the graceful DAP ~Disconnect~ request which the debug
adapter may ignore, and leaves the Spring Boot JVM running. We override
~dap-ui--update-controls~ to inject our own stop button that calls
~cfg-lsp/dap-force-quit~, which sends =SIGKILL= to the adapter process and the
JVM process group. The icon already ships with ~dap-mode~
(=icons/vscode/stop.png=).

(defun cfg-lsp/dap--kill-dap-connection (process)
  "Kill a DAP connection PROCESS and its hidden process buffer."
  (when process
    (let ((buffer (process-buffer process)))
      (ignore-errors (delete-process process))
      ;; `open-network-stream' creates buffers such as " *localhost (12345)*"
      ;; when no buffer is supplied. They are not session output and dap-mode
      ;; does not remove them when a session is cleared.
      (when (buffer-live-p buffer)
        (ignore-errors (kill-buffer buffer))))))

(defun cfg-lsp/dap-force-quit ()
  "Forcefully terminate all DAP sessions and clean up.
Kills the DAP adapter process, the terminal process (and its children)
started via runInTerminal, then clears all dap-mode state.
Safe to call even when the session is completely unresponsive."
  (interactive)
  (ignore-errors
    (dolist (session (dap--get-sessions))
      ;; Give the adapter a brief chance to release its JDWP attachment.
      (ignore-errors
        (with-timeout (1 nil)
          (dap-request session "disconnect"
                       :restart :json-false
                       :terminateDebuggee t)))
      ;; Kill the DAP connection and its hidden ` *localhost (PID)*' buffer.
      (cfg-lsp/dap--kill-dap-connection (dap--debug-session-proc session))
      ;; Kill the compile_attach program proc (Java / C++ / Node.js adapter).
      ;; compilation-start wraps the command as `sh -c "..."', so we need
      ;; pkill -P to reach the JVM child.
      (let* ((prog-proc (dap--debug-session-program-proc session))
             (prog-pid  (and prog-proc (process-id prog-proc)))
             (prog-buf  (and prog-proc (process-buffer prog-proc))))
        (when prog-pid
          (ignore-errors
            (call-process "pkill" nil nil nil "-KILL" "-P" (number-to-string prog-pid)))
          (ignore-errors
            (call-process "kill" nil nil nil "-9" (number-to-string prog-pid))))
        (ignore-errors (when prog-proc (delete-process prog-proc)))
        (ignore-errors (when (buffer-live-p prog-buf) (kill-buffer prog-buf))))
      ;; Kill the runInTerminal process (Node.js / pwa-node sessions).
      ;; start-process-shell-command also wraps in sh, so kill children too.
      (let* ((name (dap--debug-session-name session))
             (term-buf (get-buffer (format "*%s - *" name)))
             (term-buf (or term-buf
                           (cl-find-if (lambda (b)
                                         (string-prefix-p (format "*%s" name)
                                                          (buffer-name b)))
                                       (buffer-list))))
             (term-proc (and term-buf (get-buffer-process term-buf)))
             (term-pid  (and term-proc (process-id term-proc))))
        (message "cfg-lsp/dap-force-quit: term-buf=%s term-proc=%s pid=%s"
                 (and term-buf (buffer-name term-buf))
                 term-proc term-pid)
        (when term-pid
          (ignore-errors
            (call-process "pkill" nil nil nil "-KILL" "-P" (number-to-string term-pid)))
          (ignore-errors
            (call-process "kill" nil nil nil "-9" (number-to-string term-pid))))
        (ignore-errors (when term-proc (kill-process term-proc)))))
  ;; Fallback: kill any surviving dapDebugServer.js by PID (port argument uniquely
  ;; identifies this Emacs instance's adapter when program-proc tracking fails,
  ;; e.g. when stopped at a breakpoint and the process object is no longer live).
  ;; NOTE: pkill -f matches the full command line, so it only kills processes
  ;; running dapDebugServer.js -- safe as long as you don't run multiple Emacs
  ;; debug sessions concurrently.
  (ignore-errors
    (call-process "pkill" nil nil nil "-KILL" "-f" "dapDebugServer.js")))
  ;; Kill the lsp-treemacs-backed dap-ui buffers BEFORE firing session hooks.
  ;; `lsp-treemacs-wcb-unless-killed' guards every refresh with (buffer-live-p),
  ;; so killing the buffers here prevents `lsp-treemacs-generic-refresh' /
  ;; `treemacs-update-node' from being called when dap--set-sessions nil
  ;; fires dap-session-changed-hook. Without this, Treemacs refresh runs
  ;; unconditionally even when Treemacs is not open, causing background noise.
  (dolist (buf '("*dap-ui-sessions*" "*dap-ui-locals*" "*dap-ui-expressions*"))
    (when-let (b (get-buffer buf))
      (ignore-errors (kill-buffer b))))
  ;; Now it is safe to clear sessions; the hooks will find the buffers dead.
  (ignore-errors (dap--set-sessions nil))
  (ignore-errors (lsp-workspace-set-metadata "default-session" nil))
  ;; Remove orphan network buffers left after their DAP process is deleted.
  (dolist (buffer (buffer-list))
    (when (and (null (get-buffer-process buffer))
               (string-match-p "\\` \\*localhost ([0-9]+)\\*\\'"
                               (buffer-name buffer)))
      (ignore-errors (kill-buffer buffer))))
  (ignore-errors (dap-ui-hide-many-windows))
  (defvar dap-ui--control-buffer)
  (ignore-errors (posframe-hide dap-ui--control-buffer))
  (message "DAP session and processes nuked."))

(defun cfg-lsp/dap-force-stop-current-session ()
  "Force-stop the session shown by the DAP controls.
DAP controls live in their own buffer, so a buffer-local handler inherited
from a Java source buffer is not visible from a mouse click. Resolve the
session from dap-mode's global session state instead."

  (if (and (fboundp 'dap--cur-session) (dap--cur-session))
      (cfg-lsp/dap-force-quit)
    (message "No active DAP session to stop.")))

** DAP Fringe Breakpoints
We also add a convenient way to toggle breakpoints by clicking the left fringe.
This works in any buffer where ~dap-mode~ is active, allowing you to quickly set
or remove breakpoints without needing to use keyboard shortcuts or commands.

#+begin_src emacs-lisp
(declare-function treesit-fold--overlays-in "treesit-fold")
(defun cfg-lsp/dap-fringe-click-toggle-breakpoint (event)
  "Toggle a DAP breakpoint at the line clicked in the left fringe.
EVENT is the mouse click event."
  (interactive "e")
  (when (and (bound-and-true-p dap-mode)
             (derived-mode-p 'prog-mode))
    (let ((pos (posn-point (event-start event))))
      (when pos
        (save-excursion
          (goto-char pos)
          (dap-breakpoint-toggle))))))

(defun cfg-lsp/dap-treesit-fold-fringe-click-advice (orig-fn event)
  "Handle \\[mouse-1] in left-fringe in *-ts-mode for breakpoints and folds.
\\(Since the treesit-fold fringe indicator overlays everything else).
By default, clicking the fringe with no fold indicator does nothing (event is
swallowed). This advice ensures the click toggles a DAP breakpoint if no fold
indicator is present, so fringe clicks are always useful. Call ORIG-FN if a
fold indicator is present at the clicked line; otherwise, toggle a DAP
breakpoint if `dap-mode' is active in the current buffer."
  (let* ((pos (posn-point (event-start event)))
         (cur-ln (when pos (line-number-at-pos pos)))
         (ovs (when pos
                (append (treesit-fold--overlays-in
                         'type 'treesit-fold-indicators-fr-plus)
                        (treesit-fold--overlays-in
                         'type 'treesit-fold-indicators-fr-minus-tail))))
         (has-indicator
          (and ovs cur-ln
               (cl-some
                (lambda (ov)
                  (= cur-ln (line-number-at-pos (overlay-start ov))))
                ovs))))
    (if has-indicator
        ;; Fold-indicator line: if a DAP breakpoint is already there,
        ;; remove it (otherwise it would not be possible to remove
        ;; an orphaned breakpoint stuck on a fold-indicator line by
        ;; clicking on the fringe).
        (let ((bp-on-line
               (and (bound-and-true-p dap-mode)
                    pos
                    (save-excursion
                      (goto-char pos)
                      (dap--get-breakpoint-at-point)))))
          (if bp-on-line
              (cfg-lsp/dap-fringe-click-toggle-breakpoint event)
            (funcall orig-fn event)))
      (when (and (bound-and-true-p dap-mode)
                 (derived-mode-p 'prog-mode))
        (cfg-lsp/dap-fringe-click-toggle-breakpoint event)))))
