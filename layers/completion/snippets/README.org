#+TITLE: Snippets Layer
#+CATEGORY: completion
#+PROPERTY: header-args:emacs-lisp :tangle yes

* Description
Provides template-driven snippet expansion for Emacs via yasnippet and a large
collection of ready-to-use snippets across various programming languages.

* Emacs Configuration
** Yasnippet
#+begin_src emacs-lisp
(declare-function yas-reload-all "yasnippet")
(use-package yasnippet
  :demand t
  :config
  (yas-reload-all)
  (yas-global-mode 1)
  ;; After an LSP snippet expands (e.g. parseInt(${1:string})), corfu's
  ;; auto-delay timer fires 0.3 s later, reopens the popup, and corfu-map's
  ;; TAB binding wins over yas-keymap, making field navigation impossible.
  ;; We fix this scenario below by disabling corfu-auto locally for the
  ;; duration of the snippet session.
  (add-hook 'yas-before-expand-snippet-hook
            (lambda ()
              (when (fboundp 'cfg-completion/suspend-corfu-auto-for-snippet)
                (cfg-completion/suspend-corfu-auto-for-snippet))))
  (add-hook 'yas-after-exit-snippet-hook
            (lambda ()
              (when (fboundp 'cfg-completion/restore-corfu-auto-after-snippet)
                (cfg-completion/restore-corfu-auto-after-snippet)))))
#+end_src
