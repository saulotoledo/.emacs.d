#+TITLE: Corfu Layer
#+CATEGORY: completion
#+PROPERTY: header-args:emacs-lisp :tangle yes

* Description
Provides fast, lightweight in-buffer completion popups using Corfu (Completions
in Region FUnction). Leverages Emacs' native Completion At Point Functions
(CAPF) API to surface candidates from LSP/eglot, tree-sitter, keywords, dabbrev,
and snippets without blocking the editor. Includes visual icons via `kind-icon`
and additional completion backends via `cape`.

(defvar-local cfg-completion/corfu-auto-before-snippet :unset
  "Previous local `corfu-auto' value saved while a snippet is active.
The special value `:inherit' means that `corfu-auto' was not buffer-local.")

(defun cfg-completion/corfu-visible-p ()
  "Return non-nil when the Corfu popup is currently visible."
  (and (bound-and-true-p corfu-mode)
       (or (and (bound-and-true-p corfu--frame)
                (frame-live-p corfu--frame)
                (frame-visible-p corfu--frame))
           (and (featurep 'corfu-terminal)
                (fboundp 'corfu-terminal--visible-p)
                (corfu-terminal--visible-p)))))

(defun cfg-completion/corfu-candidate-selected-p ()
  "Return non-nil when Corfu has an explicitly selected candidate.
Use Corfu's own selection state; popup-frame visibility may already be false
while the active Return command is tearing the popup down."
  (and (boundp 'corfu--index)
       (integerp corfu--index)
       (>= corfu--index 0)))

(defun cfg-completion/suspend-corfu-auto-for-snippet ()
  "Suspend Corfu's automatic popup while a Yasnippet field is active."
  (when (eq cfg-completion/corfu-auto-before-snippet :unset)
    (setq-local cfg-completion/corfu-auto-before-snippet
                (if (local-variable-p 'corfu-auto)
                    corfu-auto
                  :inherit)))
  (when (boundp 'corfu-auto)
    (setq-local corfu-auto nil))
  (when (and (boundp 'corfu--auto-timer) corfu--auto-timer)
    (cancel-timer corfu--auto-timer)
    (setq corfu--auto-timer nil)))

(defun cfg-completion/restore-corfu-auto-after-snippet ()
  "Restore Corfu automatic completion after the last snippet exits."
  (when (and (not (eq cfg-completion/corfu-auto-before-snippet :unset))
             (or (not (fboundp 'yas-active-snippets))
                 (not (yas-active-snippets))))
    (if (eq cfg-completion/corfu-auto-before-snippet :inherit)
        (kill-local-variable 'corfu-auto)
      (setq-local corfu-auto cfg-completion/corfu-auto-before-snippet))
    (setq-local cfg-completion/corfu-auto-before-snippet :unset)))

(defun cfg-completion/corfu-quit-only ()
  "Dismiss Corfu without touching Copilot or ECA ghost text."
  (interactive)
  (when (fboundp 'corfu-quit)
    (corfu-quit)))

(defun cfg-completion/clear-eca-overlays ()
  "Remove ECA completion overlays at point, if ECA exposes them there."
  (dolist (overlay (overlays-at (point)))
    (when (overlay-get overlay 'completion)
      (delete-overlay overlay))))

(defun cfg-completion/corfu-return-or-newline ()
  "Accept a selected Corfu candidate or insert a newline when none is selected.
Enter is consumed while Corfu's keymap is active."
  (interactive)
  (if (cfg-completion/corfu-candidate-selected-p)
      (progn
        (when (fboundp 'copilot-clear-overlay)
          (copilot-clear-overlay))
        (cfg-completion/clear-eca-overlays)
        (corfu-insert))
    (newline)))

(defun cfg-completion/disable-company-in-corfu-buffer ()
  "Keep Company disabled whenever Corfu owns completion in this buffer.
This is an enforcement hook in addition to lsp-mode's initialization advice,
so Company cannot remain enabled or be enabled later because of hook order."
  (when (bound-and-true-p corfu-mode)
    (when (and (boundp 'company-mode)
               company-mode)
      (company-mode -1))
    ;; Do not leave an LSP-mutated backend list behind for a later Company
    ;; activation in the same buffer.
    (when (boundp 'company-backends)
      (setq-local company-backends nil))))

(defun cfg-lsp/lsp-completion-mode-without-company-for-corfu (orig-fun &rest args)
    "Prevent lsp-mode from enabling Company in Corfu buffers.
`lsp-completion-mode' still installs `lsp-completion-at-point' and the LSP
completion category settings. Only its Company autoconfiguration is skipped
when `corfu-mode' is active.
ORIG-FUN is the advised function. ARGS are its arguments."
    (if (not (bound-and-true-p corfu-mode))
        (apply orig-fun args)
      ;; Disable an already-active Company instance before LSP setup. The
      ;; Corfu layer also enforces this through company-mode-hook for later
      ;; activation; this handles the LSP initialization path itself.
      (when (and (boundp 'company-mode)
                 company-mode
                 (fboundp 'company-mode))
        (company-mode -1))
      ;; In lsp-mode 20260630, `lsp-completion-mode' installs the CAPF before
      ;; consulting `lsp-completion-provider'. Bind it to :none only for this
      ;; call so lsp-mode skips its Company branch, which otherwise enables
      ;; `company-mode' and mutates `company-backends' even in Corfu buffers.
      (let ((lsp-completion-provider :none))
        (prog1 (apply orig-fun args)
          ;; Keep the supported CAPF path explicit even if a future lsp-mode
          ;; version changes the order of its provider setup.
          (when (fboundp 'lsp-completion-at-point)
            (add-hook 'completion-at-point-functions
                      #'lsp-completion-at-point nil t))))))

(add-hook 'corfu-mode-hook #'cfg-completion/disable-company-in-corfu-buffer)

(with-eval-after-load 'company
  ;; Catch Company activation that happens after Corfu or LSP initialization.
  (add-hook 'company-mode-hook #'cfg-completion/disable-company-in-corfu-buffer)
  ;; Normalize buffers that were initialized before Company finished loading.
  (dolist (buf (buffer-list))
    (with-current-buffer buf
      (cfg-completion/disable-company-in-corfu-buffer))))

(defvar-local cfg-completion/corfu--refresh-timer nil
  "Timer used to debounce Corfu refreshes after asynchronous LSP results.")

(defun cfg-completion/corfu-refresh-after-lsp-async
    (buf &optional context-key generation saved-tick saved-point)
  "Refresh Corfu completion in BUF after an asynchronous LSP response.

BUF is the target buffer to perform completion in.
CONTEXT-KEY identifies the completion context (e.g., member access).
GENERATION is the request counter token used to detect stale responses.
SAVED-TICK is the `buffer-chars-modified-tick' at request time.
SAVED-POINT is the `point' position at request time.

If SAVED-TICK, SAVED-POINT, or GENERATION do not match the current buffer
state, the context is considered stale and the refresh is ignored.

Member-access contexts use Corfu's auto path with `.' preserved as the
trigger so an empty prefix can open the popup without lowering the global
prefix threshold. Other contexts use `completion-at-point'."
  (when-let ((win (and (buffer-live-p buf)
                       (get-buffer-window buf t))))
    (with-selected-window win
      (with-current-buffer buf
        (when (and (bound-and-true-p corfu-mode)
                   (or (not context-key)
                       (and (equal context-key cfg-lsp/lsp-completion--context-key)
                            (= generation cfg-lsp/lsp-completion--request-generation)
                            (= saved-tick (buffer-chars-modified-tick))
                            (= saved-point (point)))))
          (let ((non-essential t))
            (if (and saved-point
                     (= saved-point (point))
                     (eq (char-before saved-point) ?.)
                     (fboundp 'corfu-auto--complete-deferred))
                (let ((last-command-event ?.)
                      (this-command 'self-insert-command))
                  (corfu-auto--complete-deferred))
              (completion-at-point))))))))

(defun cfg-completion/corfu-schedule-lsp-refresh
    (buf &optional context-key generation saved-tick saved-point)
  "Schedule a Corfu refresh for an asynchronous LSP completion result.

BUF is the target buffer where completion will be refreshed.
CONTEXT-KEY identifies the completion context (e.g., member access).
GENERATION is the request counter token used to track async requests.
SAVED-TICK is the `buffer-chars-modified-tick' at request time.
SAVED-POINT is the `point' position at request time.

Schedules `cfg-completion/corfu-refresh-after-lsp-async' to run after
the LSP response returns, ensuring stale callbacks (where BUF, TICK,
POINT, or GENERATION no longer match) are ignored."
  (when (and (buffer-live-p buf)
             (with-current-buffer buf
               (and (bound-and-true-p corfu-mode)
                    (or context-key (bound-and-true-p corfu--candidates)))))
    (with-current-buffer buf
      (when (timerp cfg-completion/corfu--refresh-timer)
        (cancel-timer cfg-completion/corfu--refresh-timer))
      (setq cfg-completion/corfu--refresh-timer
            (run-at-time
             0.05 nil
             (lambda ()
               (when (buffer-live-p buf)
                 (with-current-buffer buf
                   (setq cfg-completion/corfu--refresh-timer nil)
                   (cfg-completion/corfu-refresh-after-lsp-async
                    buf context-key generation saved-tick saved-point)))))))))

(add-hook 'cfg-lsp/lsp-completion-refresh-hook
          #'cfg-completion/corfu-schedule-lsp-refresh)

(defun cfg-completion/corfu-reset-completion-cache ()
  "Clear stale LSP/Corfu completion caches and restart completion."
  (interactive)
  (when (timerp cfg-completion/corfu--refresh-timer)
    (cancel-timer cfg-completion/corfu--refresh-timer)
    (setq cfg-completion/corfu--refresh-timer nil))
  (when (and (fboundp 'corfu-reset)
             (bound-and-true-p completion-in-region-mode)
             (bound-and-true-p completion-in-region--data))
    (corfu-reset))
  (if (fboundp 'cfg-lsp/lsp-completion-clear-caches)
      (cfg-lsp/lsp-completion-clear-caches)
    (when (fboundp 'lsp-completion--clear-cache)
      (lsp-completion--clear-cache)))
  (when (fboundp 'completion-at-point)
    (let ((non-essential t))
      (completion-at-point)))
  (message "Corfu/LSP completion cache cleared"))

(use-package corfu
  :demand t
  :init
  (global-corfu-mode)
  :custom
  (corfu-auto t)
  (corfu-auto-delay 0.3)
  (corfu-auto-prefix 3)
  ;; Trigger LSP completion immediately after member-access punctuation.
  (corfu-auto-trigger ".")
  (corfu-on-exact-match 'quit)
  ;; Keep popup responsive and predictable:
  (corfu-cycle t)
  (corfu-preselect 'prompt)
  (corfu-quit-no-match t)
  ;; Reasonable popup limits:
  (corfu-scroll-margin 3)
  (corfu-count 20)
  :config
  ;; Only an explicitly selected candidate may consume Return.
  (define-key corfu-map (kbd "RET") #'cfg-completion/corfu-return-or-newline)
  (set-face-attribute 'corfu-current nil
                      :background "#3e4451"
                      :foreground "#ffffff"
                      :weight 'bold)

  ;; Disable corfu in some modes where it is noisy
  (dolist (m '(org-mode shell-mode term-mode vterm-mode))
    (add-hook (intern (concat (symbol-name m) "-hook"))
              (lambda () (corfu-mode -1))))

  ;; `corfu-auto--complete-deferred` fires the capf chain directly via
  ;; run-hook-wrapped, bypassing `corfu--exhibit` which is the only place
  ;; corfu binds `(non-essential t)`. Without this, our lsp completion
  ;; debounce (and lsp-mode itself) cannot distinguish auto-completion
  ;; background requests from interactive ones.
  (with-eval-after-load 'corfu-auto
    (advice-add 'corfu-auto--complete-deferred :around
                (lambda (orig-fun &rest args)
                  (let ((non-essential t))
                    (apply orig-fun args))))))

option in the list.

#+begin_src emacs-lisp

  :after corfu
  :custom

** CAPE (Completion At Point Extensions)
CAPE provides CAPF-based replacements for common Company backends, exposing file
names, buffer words, and language keywords through Emacs' standard
~completion-at-point-functions~ API. This allows Corfu (and other CAPF
frontends) to consume these sources natively, without Company-specific backends
or blocking behavior.

#+begin_src emacs-lisp
(use-package cape
  :init
  (add-to-list 'completion-at-point-functions #'cape-file)
  (add-to-list 'completion-at-point-functions #'cape-dabbrev)
  (add-to-list 'completion-at-point-functions #'cape-keyword))
#+end_src
