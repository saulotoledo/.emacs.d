* Description
Integrates the ultra-fast Ripgrep search tool with Emacs. Provides buffer
search, project-wide text searching via transient menus, and auto-saving
editable grep results via wgrep.

* Emacs Configuration
** Ripgrep Package Setup
#+begin_src emacs-lisp
(use-package rg
  :config
  (rg-enable-default-bindings)
  (rg-enable-menu)
  (setq wgrep-auto-save-buffer t) ; Save wgrep buffers automatically after editing

  (defun my/counsel-rg-auto-preview-advice (orig-fun &rest args)
    "Enable live-calling/preview automatically when running `counsel-rg`."
    (minibuffer-with-setup-hook
        (lambda () (setq-local ivy-calling t))
      (apply orig-fun args)))

  (advice-add 'counsel-rg :around #'my/counsel-rg-auto-preview-advice)

  (declare-function project-roots "project")
  (declare-function transient-quit-all "transient")
  (defun cfg-tools/rg-abort-to-counsel ()
    "Abort the current transient menu and launch counsel-rg."
    (interactive)
    (transient-quit-all)
    (cfg-tools/counsel-rg-dir))

  (declare-function transient-suffix-object "transient")
  (declare-function transient-lisp-variable "transient")
  (declare-function rg-menu-transient-insert "rg")
  (transient-define-infix cfg-tools/search-case-strategy ()
    :class 'transient-lisp-variable
    :variable 'rg-ignore-case
    :description
    (lambda ()
      (format "Search Case Strategy [%s]" (or rg-ignore-case "nil")))
    :reader
    (lambda (_prompt _initial _history)
      (let* ((choices '(case-fold-search smart force nil))
             (pos (or (cl-position rg-ignore-case choices) 0)))
        (nth (mod (1+ pos) (length choices)) choices))))

  (rg-menu-transient-insert "Search" "z" "Live Search (Counsel)" 'cfg-tools/rg-abort-to-counsel)
  (rg-menu-transient-insert "Custom" "g" "Search Case Strategy" 'cfg-tools/search-case-strategy)

  (add-to-list 'display-buffer-alist
               '("\\*rg\\*"
                 (display-buffer-same-window))))

(defun cfg-tools/counsel-rg (&optional prompt-dir hidden)
  "Run `counsel-rg'.
If PROMPT-DIR is non-nil (or called interactively with a prefix arg \\[universal-argument]),
prompt for the directory instead of defaulting to project root.
If HIDDEN is non-nil, search hidden and ignored files/folders."
  (interactive "P")
  (let* ((proj-root
          (cond
           ((fboundp 'projectile-project-root)
            (projectile-project-root))
           ((fboundp 'project-current)
            (let ((proj (project-current)))
              (when proj
                (if (fboundp 'project-root)
                    (project-root proj)
                  (car (project-roots proj))))))
           ((fboundp 'vc-root-dir)
            (vc-root-dir))
           (t default-directory)))
         (dir (if prompt-dir
                  (read-directory-name "Search directory: " proj-root nil t)
                proj-root))
         (counsel-rg-base-command
          (if hidden
              "rg -M 240 --with-filename --no-heading --line-number --color never --hidden --no-ignore %s"
            (or (bound-and-true-p counsel-rg-base-command)
                "rg -M 240 --with-filename --no-heading --line-number --color never %s"))))
    ;; Call counsel-rg inside the let* block so counsel-rg-base-command is properly scoped
    (counsel-rg nil dir)))

(defun cfg-tools/counsel-rg-dir ()
  "Run counsel-rg with directory prompt."
  (interactive)
  (cfg-tools/counsel-rg t nil))

(defun cfg-tools/counsel-rg-hidden ()
  "Run counsel-rg with hidden files enabled."
  (interactive)
  (cfg-tools/counsel-rg nil t))

(defun cfg-tools/counsel-rg-dir-hidden ()
  "Run counsel-rg with directory prompt AND hidden files enabled."
  (interactive)
  (cfg-tools/counsel-rg t t))
#+end_src

is_installed__tools__rg__ripgrep() {
  is_system_package_installed "ripgrep"
}

register_system_package "ripgrep" \
                        "alpine:ripgrep" \
                        "fedora:ripgrep" \
                        "linuxbrew:ripgrep" \
                        "mac:ripgrep" \
                        "debian-ubuntu:ripgrep"

install__tools__rg__ripgrep() {
  log_info "Starting Ripgrep setup..."
  install_system_packages "ripgrep"
}

uninstall__tools__rg__ripgrep() {
  log_skip "Ripgrep" "Ripgrep is system package and will not be uninstalled automatically"
}
