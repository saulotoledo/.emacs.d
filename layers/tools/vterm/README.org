* Description
Provides full terminal emulation inside Emacs using vterm compiled C module.
Alternatives like ~Eshell~ or ~Term~ are written in Elisp; while they integrate
deeply with Emacs, they often lag, flicker, or break entirely when running
resource-heavy programs like =htop= or =vim=. By using ~vterm~, we get a "real"
terminal experience that handles complex character encoding and high-speed
scrolling without the compromises found in pure Emacs Lisp shells.

(defun cfg-tools/vterm-kill-with-no-query-around (orig-fun &rest args)
  "Do not ask about killing vterm buffers.
It uses `advice-add' instead of defadvice.
ORIG-FUN is the advised function and ARGS are its arguments."
  (let ((buf (apply orig-fun args)))
    (when (and (bufferp buf) (get-buffer-process buf))
      (set-process-query-on-exit-flag (get-buffer-process buf) nil))
    buf))

(defun cfg-tools/toggle-vterm (&optional n)
  "Toggle between a `vterm' buffer and the previous non-vterm buffer.
Creates a new `vterm' rooted at `vc-root-dir' (or `default-directory')
if none exist or if numeric argument N is provided."
  (interactive)
  (setq cfg-tools/vterms (seq-filter 'buffer-live-p cfg-tools/vterms))
  (let ((default-directory (or (vc-root-dir) default-directory)))
    (cond ((numberp n) (push (vterm n) cfg-tools/vterms))
          ((null cfg-tools/vterms) (push (vterm 1) cfg-tools/vterms))
          ((seq-contains-p cfg-tools/vterms (current-buffer))
           (switch-to-buffer (car (seq-difference (buffer-list) cfg-tools/vterms))))
          (t (switch-to-buffer (car (seq-intersection (buffer-list) cfg-tools/vterms)))))))

(defun cfg-tools/vterm-send-next-key ()
  "Prompt for a key sequence and send it to the vterm terminal."
  (interactive)
  (when (derived-mode-p 'vterm-mode)
    (message "The next key will be sent to vterm")
    (vterm-send-next-key)))

(defun cfg-tools/vterm-send-current-key ()
  "Send the key sequence that invoked this command directly to vterm."
  (interactive)
  (vterm-send-key (key-description (this-command-keys))))

(defun cfg-tools/vterm-setup-key-remaps ()
  "Remap mwim commands to send native keys in vterm buffers."
  (local-set-key [remap mwim-beginning-of-code-or-line] #'cfg-tools/vterm-send-current-key)
  (local-set-key [remap mwim-end-of-code-or-line]       #'cfg-tools/vterm-send-current-key))

(defun cfg-tools/terminal-disable-editor-modes ()
  "Disable editor-only modes in a terminal buffer."
  (dolist (mode '(apheleia-mode
                  auto-composition-mode
                  display-fill-column-indicator-mode
                  flycheck-annotate-mode
                  show-smartparens-mode
                  visual-line-mode))
    (when (fboundp mode)
      (funcall mode -1)))
  (setq-local indent-tabs-mode nil)
  ;; `override-global-mode' is global; a local binding keeps it disabled in
  ;; this terminal without changing keybindings in other buffers.
  (setq-local override-global-mode nil))

(defun cfg-tools/vterm-mouse-open-at-point (event)
  "Open URL or file at mouse EVENT in `vterm-mode', or pass click to `vterm'."
  (interactive "e")
  (require 'ffap)
  (let* ((posn (event-start event))
         (win (posn-window posn))
         (pt (posn-point posn))
         (target-type nil)
         (target nil))
    (when (and (windowp win) (integerp pt))
      (with-selected-window win
        (save-excursion
          (goto-char pt)
          (cond
           ((setq target (ffap-url-at-point))
            (setq target-type 'url))
           ((setq target (ffap-file-at-point))
            (setq target-type 'file))))))
    (pcase target-type
      ('url
       (browse-url target))
      ('file
       (find-file-other-window target))
      (_
       ;; Not a link: fall back to standard vterm click handling
       (if (fboundp 'vterm--mouse-set-point)
           (vterm--mouse-set-point event)
         (mouse-set-point event))))))

(defun cfg-tools/vterm-copy-mode-open-at-point ()
  "Open the URL or file path under point."
  (interactive)
  (let* ((url (ffap-url-at-point))
         (file (and (not url) (ffap-file-at-point))))
    (cond
     (url
      (browse-url url))
     (file
      (find-file-other-window file))
     (t
      (message "No URL or valid file path found at point")))))

(defun cfg-tools/vterm-copy-mode-mouse-open (event)
  "Move point to the mouse click EVENT position and open the link at point."
  (interactive "e")
  (mouse-set-point event)
  (cfg-tools/vterm-copy-mode-open-at-point))

* System Setup
~vterm~ requires a few development packages installed on your system to allow
Emacs to build the driver during the initial setup:

#+begin_src sh
VTERM_BUILD_PACKAGES=(
  "libvterm" "cmake" "build-essential"
)

is_installed__tools__vterm__vterm_build_bundle() {
  are_these_system_packages_installed "${VTERM_BUILD_PACKAGES[@]}"
}

register_system_package "libvterm" \
                        "alpine:libvterm-dev" \
                        "fedora:libvterm-devel" \
                        "linuxbrew:libvterm" \
                        "mac:libvterm" \
                        "debian-ubuntu:libvterm-dev"

register_system_package "cmake" \
                        "alpine:cmake" \
                        "fedora:cmake" \
                        "linuxbrew:cmake" \
                        "mac:cmake" \
                        "debian-ubuntu:cmake"

register_system_package "build-essential" \
                        "alpine:build-base" \
                        "fedora:gcc-c++" \
                        "linuxbrew:gcc" \
                        "mac:xcode" \
                        "debian-ubuntu:build-essential"

install__tools__vterm__vterm_build_bundle() {
  log_info "Starting C++ Development environment setup..."
  install_system_packages "${VTERM_BUILD_PACKAGES[@]}"
}

uninstall__tools__vterm__vterm_build_bundle() {
  log_skip "vterm build bundle" "The packages used for vterm builds are system packages and will not be uninstalled automatically"
}

register_action "Tools" "VTerm" "Install vterm build bundle" "-tools--vterm--vterm_build_bundle" "utils"
#+end_src
