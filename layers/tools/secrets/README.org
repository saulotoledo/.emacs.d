#+TITLE: Secrets Layer
#+CATEGORY: tools
#+PROPERTY: header-args:emacs-lisp :tangle yes

* Description
A security-focused layer utilizing Cloak Mode to automatically mask sensitive
data (such as API keys, tokens, and passwords) with asterisks in `.env`, `yaml`,
`json`, and password files. Prevents accidental exposure during screen shares
and provides a safe prompt for inline editing.

* Emacs Configuration
** Cloak Mode Secrets Masking
#+begin_src emacs-lisp
(defvar cfg-tools/cloak-keywords '("KEY" "PASSWD" "PASS" "PASSWORD" "SECRET" "TOKEN" "AUTH")
  "List of keywords that trigger cloaking.")

(defun cfg-tools/get-cloak-regex (sep)
  "Regex for cloak-mode. SEP is the separator between key and value."
  (let ((kw (regexp-opt cfg-tools/cloak-keywords)))
    (concat "^[ \t]*[^" sep "\n]*" kw "[^" sep "\n]*"
            sep "[ \t]*[\"']?\\(.*?\\)[\"']?$")))

(defconst cfg-tools/cloak-regex-dot-env (cfg-tools/get-cloak-regex "="))
(defconst cfg-tools/cloak-regex-yaml (cfg-tools/get-cloak-regex ":"))
(defconst cfg-tools/cloak-regex-whole-line "^\\(.+\\)$")

(defun cfg-tools/cloak-passwd-file-p ()
  "Return t if the filename has \"passwd\" or \"password\" and only 1 non-blank line."
  (let ((filename (buffer-file-name)))
    (and filename
         (string-match-p "passw\\(d\\|ord\\)" filename)
         (save-excursion
           (save-match-data
             (goto-char (point-min))
             (if (re-search-forward "[^ \t\n]" nil t)
                 (progn
                   (forward-line 1)
                   (not (re-search-forward "[^ \t\n]" nil t)))
               t))))))

(define-derived-mode cfg-tools/passwd-mode fundamental-mode "Passwd"
  "Major mode for single-line password files to hide their content.")

(add-to-list 'magic-mode-alist (cons #'cfg-tools/cloak-passwd-file-p #'cfg-tools/passwd-mode))

(use-package cloak-mode
  :config
  (setq cloak-mode-patterns
        `((cfg-tools/passwd-mode . ,cfg-tools/cloak-regex-whole-line)
          (dotenv-mode           . ,cfg-tools/cloak-regex-dot-env)
          (yaml-mode             . ,cfg-tools/cloak-regex-yaml)
          (json-mode             . ,cfg-tools/cloak-regex-yaml)))

  (defun cfg-tools/cloak-ensure-on ()
    "Force cloak-mode on if the filename matches secret criteria."
    (when (cfg-tools/cloak-passwd-file-p)
      (cloak-mode 1)))

  (add-hook 'find-file-hook 'cfg-tools/cloak-ensure-on)
  (add-hook 'after-change-major-mode-hook 'cfg-tools/cloak-ensure-on)
  (add-hook 'cfg-tools/passwd-mode-hook 'cfg-tools/cloak-ensure-on)

  (defun cfg-tools/cloak-edit ()
    "Edit the secret value at point."
    (interactive)
    (let* ((is-pass-file (cfg-tools/cloak-passwd-file-p))
           (sep (if (derived-mode-p 'yaml-mode 'json-mode) ":" "="))
           (regex (if is-pass-file cfg-tools/cloak-regex-whole-line (cfg-tools/get-cloak-regex sep))))
      (save-excursion
        (beginning-of-line)
        (if (re-search-forward regex (line-end-position) t)
            (let* ((old-val (match-string 1))
                   (beg (match-beginning 1))
                   (end (match-end 1))
                   (new-val (read-string "New Secret Value: " old-val)))
              (let ((inhibit-read-only t))
                (delete-region beg end)
                (goto-char beg)
                (insert new-val))
              (when cloak-mode
                (cloak-mode -1)
                (cloak-mode 1))
              (message "Secret updated."))
          (message "No secret found on this line.")))))

  (global-cloak-mode 1))
#+end_src
