* Description
Provides editing support and language server integration for Terraform and
HashiCorp Configuration Language (HCL). Enables syntax highlighting,
auto-formatting, and LSP features like auto-completion and diagnostics for
managing Infrastructure as Code (IaC).

* Emacs Configuration
** Terraform Mode
#+begin_src emacs-lisp
(use-package terraform-mode
  :defer t
  :custom
  (terraform-command (if (executable-find "tofu") "tofu" "terraform"))
  :hook ((terraform-mode . lsp-deferred)
         (terraform-mode . terraform-format-on-save-mode)))
#+end_src

** LSP language ID
~terraform-mode~ handles both ~*.tf~ and ~*.tfvars~, but terraform-ls expects
different language IDs: ~terraform~ for config files and ~terraform-vars~ for
variable files. Without the mapping below, every assignment in ~*.tfvars~ is
validated as a module attribute and reported as "Unexpected attribute".

#+begin_src emacs-lisp
(with-eval-after-load 'lsp-mode
  ;; Must precede (terraform-mode . "terraform") in lsp-language-id-configuration.
  ;; add-to-list inserts at front, so filename regex wins for *.tfvars buffers.
  (add-to-list 'lsp-language-id-configuration '("\\.tfvars\\'" . "terraform-vars"))

  (when (boundp 'cfg-lsp/server-file-extensions)
    (add-to-list 'cfg-lsp/server-file-extensions '(tfmls . ("tf" "tfvars")))))
#+end_src

** Format on save
On save, ~terraform-format-on-save-mode~ formats the buffer with ~fmt~. That is
the same function as ~terraform-format-buffer~.

The binary is ~terraform-command~:

1. ~tofu~ if it is on PATH (OpenTofu)
2. otherwise ~terraform~ (HashiCorp)

Apheleia is the usual on-save formatter in this config, but it is a poor fit
here:
- It is loaded lazily, so opening a ~.tf~ file does not enable it.
- Its default for ~terraform-mode~ is OpenTofu (~tofu fmt~), which fails
  silently when ~tofu~ is missing.
- ~terraform-mode~ derives from ~hcl-mode~, so Apheleia can also try ~hclfmt~.

Disable Apheleia for Terraform so save uses only the path above.

#+begin_src emacs-lisp
(defvar apheleia-mode-alist nil)

(with-eval-after-load 'apheleia
  (setf (alist-get 'terraform-mode apheleia-mode-alist) nil))
#+end_src

(cfg-keys/mode-leader-def terraform-mode

* System Setup
We need to install the Terraform Language Server on our system:

#+begin_src sh
is_installed__cloud__terraform__terraformls() {
  is_system_package_installed "terraformls"
}

register_system_package "terraformls" \
                        "alpine:terraform-ls" \
                        "fedora:terraform-ls" \
                        "linuxbrew:hashicorp/tap/terraform-ls" \
                        "mac:hashicorp/tap/terraform-ls" \
                        "debian-ubuntu:terraform-ls"

install__cloud__terraform__terraformls() {
  log_info "Starting Terraform Language Server..."

  if [[ "$OSTYPE" == "darwin"* ]]; then
    brew tap hashicorp/tap
  elif [ -f /etc/debian_version ]; then
    if [ ! -f /usr/share/keyrings/hashicorp-archive-keyring.gpg ] || [[ "$FORCE_REINSTALL" == "true" ]]; then
      wget -O- https://apt.releases.hashicorp.com/gpg | sudo gpg --dearmor --yes -o /usr/share/keyrings/hashicorp-archive-keyring.gpg
    fi
    if [ ! -f /etc/apt/sources.list.d/hashicorp.list ] || [[ "$FORCE_REINSTALL" == "true" ]]; then
      echo "deb [signed-by=/usr/share/keyrings/hashicorp-archive-keyring.gpg] https://apt.releases.hashicorp.com $(lsb_release -cs) main" | sudo tee /etc/apt/sources.list.d/hashicorp.list > /dev/null
    fi
  elif [ -f /etc/fedora-release ] || command -v dnf >/dev/null 2>&1; then
    if [ ! -f /etc/yum.repos.d/hashicorp.repo ] || [[ "$FORCE_REINSTALL" == "true" ]]; then
      sudo dnf config-manager addrepo --from-repofile=https://rpm.releases.hashicorp.com/fedora/hashicorp.repo --overwrite
    fi
  elif [ -f /etc/alpine-release ] || command -v apk >/dev/null 2>&1; then
    # We need to enable the comunity repo on Alpine
    VERSION=$(cut -d. -f1,2 /etc/alpine-release)
    REPO_URL="https://dl-cdn.alpinelinux.org/alpine/v${VERSION}/community"

    if ! grep -q "$REPO_URL" /etc/apk/repositories; then
      echo "$REPO_URL" >> /etc/apk/repositories
      apk update
    fi
  elif command -v brew >/dev/null 2>&1; then
    brew tap hashicorp/tap
  else
    log_error "OS not supported for auto-installation."
    return 1
  fi

  install_system_packages "terraformls"
}

uninstall__cloud__terraform__terraformls() {
  log_skip "Terraform Language Server" "Terraform Language Server is system package and will not be uninstalled automatically"
}

register_action "Cloud" "Terraform" "Install Terraform Language Server" "-cloud--terraform--terraformls" "cloud"
#+end_src
